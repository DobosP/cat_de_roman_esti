"""Independent final factual artifact validation, retaining the semantic review identity."""
from pathlib import Path
import hashlib
import json
import sys

ROOT=Path('/home/dobo/work/_worktrees/cat_de_roman_esti/feat__v95-discovery-and-game-quality')
OUT=Path(__file__).resolve().parent
sys.path.insert(0,str(ROOT))
from scripts import build_quick_content_v92 as B
from cat_de_roman_esti.wordgames.derived_catalog import load_derived_catalog
from cat_de_roman_esti.wordgames.service import get_service
Q=B.Q
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def read(p):return json.loads(p.read_bytes())
def write(name,value):(OUT/name).write_bytes(Q.json_bytes(value))
current=B.build_catalog(OUT/'candidate.json',OUT/'factual-review.json',OUT/'quality-review.json')
assert Q.json_bytes(current)==(OUT/'catalog.json').read_bytes()
assert Q.CATALOG_SHA256==sha(OUT/'catalog.json')
factual=read(OUT/'factual-review.json');audit=read(OUT/'live-audit.json')
assert factual['reviewer']=='/root/v95_world_author'
assert audit['kind']==B.AUDIT_KIND and audit['passed']
assert audit['catalog_sha256']==sha(OUT/'catalog.json')
assert audit['candidate_sha256']==sha(OUT/'candidate.json')
assert audit['bindings']==Q.bindings(live=True)==current['bindings']
assert audit['runtime_sources']==B.runtime_source_hashes()
assert audit['core_boards_preserved']
assert audit['base_counts']=={'intrusul':183,'perechi':153}
assert audit['combined_counts']=={'intrusul':226,'perechi':191}
assert len(audit['replays'])==len(current['boards'])==len(current['authored'])==81
assert {r['id'] for r in audit['replays']}=={r['id'] for r in current['boards']}
assert all(r['score']==1000 and r['hidden_answers_preserved'] and r['terminal_get_preserved'] for r in audit['replays'])
previous=read(ROOT/'docs/reviews/v94-words-and-clearer-connections/quick/catalog.json')
assert current['authored'][:73]==previous['authored']
for before,after in zip(previous['boards'],current['boards'][:73],strict=True):
    assert {k:v for k,v in before.items() if k not in ('standard_rank','starter_rank')}=={k:v for k,v in after.items() if k not in ('standard_rank','starter_rank')}
base=load_derived_catalog();Q.validate_catalog(current,base,get_service())
evidence={'kind':'quick-independent-final-factual-check-v1','reviewer':factual['reviewer'],
    'candidate_sha256':sha(OUT/'candidate.json'),'catalog_sha256':sha(OUT/'catalog.json'),
    'audit_sha256':sha(OUT/'live-audit.json'),'factual_review_sha256':sha(OUT/'factual-review.json'),
    'runtime_sources':B.runtime_source_hashes(),'bindings':Q.bindings(live=True),
    'natural_public_seed_wins':81,'core_boards_exact':336,'prior_authored_records_exact':73,
    'prior_payload_scores_exact_except_competition_ranks':True,'checked_new_factual_rows':8,
    'combined_counts':audit['combined_counts'],'passed':True}
write('independent-final-factual-check.json',evidence)
write('final-factual.json',{'kind':B.FINAL_KIND,'role':'factual','reviewer':factual['reviewer'],
    'catalog_sha256':sha(OUT/'catalog.json'),'audit_sha256':sha(OUT/'live-audit.json'),
    'verdict':'accept','rationale':'The exact 81-board proposal reproduces the independently verified candidate and both complete semantic reviews. All eight new ordinary meanings and source-backed predicates remain exact; 73 prior records, payloads and non-ranking scores are preserved. Current graph/rubric/pack/derived and runtime bindings match the final live audit. All 81 authored boards naturally select and win through the API with hidden identifiers/answers and terminal GET preservation, while all 336 core boards remain intact. This acceptance concerns content integrity and factual meanings, not measured enjoyment.',
    'independent_check_sha256':sha(OUT/'independent-final-factual-check.json')})
print(json.dumps({'final_factual_sha256':sha(OUT/'final-factual.json'),'catalog_sha256':sha(OUT/'catalog.json'),'audit_sha256':sha(OUT/'live-audit.json')}))
