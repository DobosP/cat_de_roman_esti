from pathlib import Path

from cat_de_roman_esti.graph import Graph
from cat_de_roman_esti.wordgames import alchimie
from cat_de_roman_esti.wordgames.packs import (
    _closure_generations,
    _opening_pairs,
    lant_branch_profile,
    minimum_alchimie_actions,
    validate_payload,
)

ROOT = Path('/home/dobo/work/_worktrees/cat_de_roman_esti/fix__v81-nuca-feedback')
SCRATCH = Path('/home/dobo/work/_temp/v81-nuca-feedback/impact')
BASE = 'dbbcf6e8b810fb990c9acffb42d6426138551637'
KG_PATH = ROOT / 'cat_de_roman_esti/fixtures/kg_sample.json'
PACK_PATH = ROOT / 'cat_de_roman_esti/fixtures/games_pack.json'
RANK_PATH = ROOT / 'cat_de_roman_esti/fixtures/board_rankings_v37.json'
CATALOG_PATH = ROOT / 'cat_de_roman_esti/fixtures/derived_catalog_v38.json'
MOBILE_PATH = ROOT / 'tests/fixtures/cat_mobile_app_pack_contract.json'
NUCA = 'n_v81_food_pantry_nuca'
TARGETS = ('n_gas_cozonac', 'n_v17gas_coliva', 'n_gas_baclava_dobrogeana', 'n_v21gas_cornulete')
IDS = {NUCA, *TARGETS}


def projection(svc, rec):
    ref = alchimie._ProjectionServiceRef(svc)
    value = alchimie._build_recipe_projection_cached(
        tuple(rec['seeds']), rec['target'], rec.get('category'), ref
    )
    if value is None:
        return None
    return {
        'par': value.par,
        'recipes': [
            [list(pair), list(outputs)] for pair, outputs in sorted(value.recipes.items())
        ],
        'routes': [
            [[list(pair), list(outputs)] for pair, outputs in route]
            for route in value.routes
        ],
    }


def show_json(path: Path):
    relative = path.relative_to(ROOT)
    raw = subprocess.check_output(['git', 'show', f'{BASE}:{relative}'], cwd=ROOT)
    return json.loads(raw)


def make_service(kg):
    return WordGameService(Graph.from_records(kg['kg_nodes'], kg['kg_edges']))


def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    SCRATCH.mkdir(parents=True, exist_ok=True)
    pack = json.loads(PACK_PATH.read_text(encoding='utf-8'))
    baseline_pack = show_json(PACK_PATH)
    kg = show_json(KG_PATH)
    prospective = json.loads(KG_PATH.read_text(encoding='utf-8'))
    baseline_svc = make_service(kg)
    prospective_svc = make_service(prospective)
    report = {
        'schema': 'v81-actual-all-six-impact-v1',
        'valid_until': 'V81 fixture, game pack, ranking, derived catalog, or runtime graph logic changes',
        'bindings': {
            'baseline_commit': BASE,
            'current_head': subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip(),
            'git_status': subprocess.check_output(['git', 'status', '--short'], cwd=ROOT, text=True).splitlines(),
            'current_hashes': {str(path.relative_to(ROOT)): sha256(path) for path in (KG_PATH, PACK_PATH, RANK_PATH, CATALOG_PATH, MOBILE_PATH)},
        },
        'pack_exact': pack == baseline_pack,
    }

    game_kinds = ('conexiuni', 'contexto', 'lant', 'alchimie')
    report['pack_counts'] = {game: len(pack[game]) for game in game_kinds}
    report['occurrences'] = []
    for game in game_kinds:
        records = pack[game]
        for rec in records:
            touched = []
            if game == 'conexiuni':
                for group, members in rec['groups'].items():
                    for nid in IDS.intersection(members):
                        touched.append({'id': nid, 'role': f'group:{group}'})
            else:
                for field in ('target', 'start'):
                    if rec.get(field) in IDS:
                        touched.append({'id': rec[field], 'role': field})
                for nid in IDS.intersection(rec.get('seeds', [])):
                    touched.append({'id': nid, 'role': 'seed'})
            if touched:
                report['occurrences'].append({
                    'game': game, 'record': rec['id'], 'status': rec['status'],
                    'difficulty': rec['difficulty'], 'touched': touched,
                })

    report['validation'] = []
    for game in game_kinds:
        records = pack[game]
        for rec in records:
            before = validate_payload(rec, game, baseline_svc)
            after = validate_payload(rec, game, prospective_svc)
            if before != after:
                report['validation'].append({
                    'game': game, 'record': rec['id'], 'status': rec['status'],
                    'before': before, 'after': after,
                })

    report['contexto'] = []
    for rec in pack['contexto']:
        target = rec['target']
        bd, ad = baseline_svc.distances_to(target), prospective_svc.distances_to(target)
        bw = baseline_svc.weighted_distances_to(target)
        aw = prospective_svc.weighted_distances_to(target)
        changed_nodes = sorted(
            nid for nid in set(bd) | set(ad) | set(bw) | set(aw)
            if bd.get(nid) != ad.get(nid) or bw.get(nid) != aw.get(nid)
        )
        if changed_nodes:
            report['contexto'].append({
                'record': rec['id'], 'status': rec['status'], 'target': target,
                'reachable': [len(bd), len(ad)],
                'responsive': [sum(1 <= d <= 5 for d in bd.values()), sum(1 <= d <= 5 for d in ad.values())],
                'nuca_hops': [bd.get(NUCA), ad.get(NUCA)],
                'nuca_weighted': [bw.get(NUCA), aw.get(NUCA)],
                'changed_node_count': len(changed_nodes),
                'changed_nodes_sample': changed_nodes[:20],
            })

    report['lant'] = []
    for rec in pack['lant']:
        before_dist = baseline_svc.distance(rec['start'], rec['target'])
        after_dist = prospective_svc.distance(rec['start'], rec['target'])
        before_profile = lant_branch_profile(baseline_svc, rec['start'], rec['target'], before_dist) if before_dist is not None else None
        after_profile = lant_branch_profile(prospective_svc, rec['start'], rec['target'], after_dist) if after_dist is not None else None
        before_paths = critique_pack.representative_shortest_paths(baseline_svc, rec['start'], rec['target'], before_dist) if before_dist else []
        after_paths = critique_pack.representative_shortest_paths(prospective_svc, rec['start'], rec['target'], after_dist) if after_dist else []
        if (before_dist, before_profile, before_paths) != (after_dist, after_profile, after_paths):
            report['lant'].append({
                'record': rec['id'], 'status': rec['status'], 'start': rec['start'],
                'target': rec['target'], 'stored': rec['optimal'],
                'distance': [before_dist, after_dist],
                'profile': [before_profile, after_profile],
                'before_paths': before_paths, 'after_paths': after_paths,
            })

    report['alchimie'] = []
    for index, rec in enumerate(pack['alchimie'], 1):
        category = rec.get('category')
        before_closure = _closure_generations(baseline_svc, rec['seeds'], category)
        after_closure = _closure_generations(prospective_svc, rec['seeds'], category)
        before_actions = minimum_alchimie_actions(baseline_svc, rec['seeds'], rec['target'], category)
        after_actions = minimum_alchimie_actions(prospective_svc, rec['seeds'], rec['target'], category)
        before_open = _opening_pairs(baseline_svc, rec['seeds'], category)
        after_open = _opening_pairs(prospective_svc, rec['seeds'], category)
        before_recipe = critique_pack.minimum_alchimie_recipe(baseline_svc, rec['seeds'], rec['target'], category, 6)
        after_recipe = critique_pack.minimum_alchimie_recipe(prospective_svc, rec['seeds'], rec['target'], category, 6)
        before_projection = projection(baseline_svc, rec)
        after_projection = projection(prospective_svc, rec)
        changed = sorted(nid for nid in set(before_closure) | set(after_closure) if before_closure.get(nid) != after_closure.get(nid))
        if changed or (before_actions, before_open, before_recipe, before_projection) != (after_actions, after_open, after_recipe, after_projection):
            report['alchimie'].append({
                'record': rec['id'], 'status': rec['status'], 'target': rec['target'],
                'seeds': rec['seeds'], 'depth': rec['target_depth'],
                'closure_size': [len(before_closure), len(after_closure)],
                'changed_closure_count': len(changed), 'changed_closure_sample': changed[:30],
                'actions': [before_actions, after_actions], 'openings': [before_open, after_open],
                'minimum_recipe_changed': before_recipe != after_recipe,
                'projection_changed': before_projection != after_projection,
                'before_minimum_recipe': before_recipe, 'after_minimum_recipe': after_recipe,
                'before_projection': before_projection, 'after_projection': after_projection,
            })

    report['conexiuni'] = []
    for rec in pack['conexiuni']:
        members = {nid for ids in rec['groups'].values() for nid in ids}
        node_types = {nid: baseline_svc.node(nid).node_type for nid in members}
        bn = {nid: set(baseline_svc.neighbor_ids(nid)) for nid in members}
        an = {nid: set(prospective_svc.neighbor_ids(nid)) for nid in members}
        bf = critique_pack.fairness_counts(rec['groups'], bn, node_types)
        af = critique_pack.fairness_counts(rec['groups'], an, node_types)
        group_for = {nid: group for group, ids in rec['groups'].items() for nid in ids}
        new_cross = []
        if bf != af or new_cross:
            report['conexiuni'].append({
                'record': rec['id'], 'status': rec['status'], 'fairness': [bf, af],
                'new_symmetric_critique_cross_edges': new_cross,
            })

    source_ids = derived._V38_SOURCE_SNAPSHOT
    records = [r for r in pack['conexiuni'] if r['id'] in source_ids]
    def derive_for(graph_kg, svc):
        strengths, linked = derived._edge_maps(graph_kg)
        raw_i = derived._intrusul_candidates(records, svc, strengths, linked)
        raw_p = derived._perechi_candidates(records, svc, strengths)
        rows = derived._rank_and_cap([*raw_i, *raw_p])
        return raw_i, raw_p, rows
    bi, bp, brows = derive_for(kg, baseline_svc)
    ai, ap, arows = derive_for(prospective, prospective_svc)
    by_id_before = {r['id']: r for r in brows}
    by_id_after = {r['id']: r for r in arows}
    changed_common = sorted(i for i in by_id_before.keys() & by_id_after if by_id_before[i] != by_id_after[i])
    report['derived'] = {
        'raw_counts': [[len(bi), len(bp)], [len(ai), len(ap)]],
        'selected_counts': [len(brows), len(arows)],
        'selected_by_game': [
            {g: sum(r['game'] == g for r in rows) for g in derived.DERIVED_GAMES}
            for rows in (brows, arows)
        ],
        'removed_ids': sorted(by_id_before.keys() - by_id_after.keys()),
        'added_ids': sorted(by_id_after.keys() - by_id_before.keys()),
        'changed_common_count': len(changed_common),
        'changed_common': [
            {'id': i, 'source': by_id_before[i]['source_id'], 'game': by_id_before[i]['game'],
             'before_rank': [by_id_before[i]['standard_rank'], by_id_before[i]['starter_rank']],
             'after_rank': [by_id_after[i]['standard_rank'], by_id_after[i]['starter_rank']]}
            for i in changed_common
        ],
        'removed': [by_id_before[i] for i in sorted(by_id_before.keys() - by_id_after.keys())],
        'added': [by_id_after[i] for i in sorted(by_id_after.keys() - by_id_before.keys())],
    }
    existing = json.loads(CATALOG_PATH.read_text(encoding='utf-8'))['boards']
    baseline_existing = show_json(CATALOG_PATH)['boards']
    report['derived']['baseline_equals_frozen_boards'] = brows == existing
    report['derived']['baseline_rebuild_equals_baseline_frozen_boards'] = brows == baseline_existing
    report['derived']['current_rebuild_equals_current_frozen_boards'] = arows == existing

    baseline_rank = show_json(RANK_PATH)
    current_rank = json.loads(RANK_PATH.read_text(encoding='utf-8'))
    baseline_rank_by_id = {row['id']: row for row in baseline_rank['boards']}
    current_rank_by_id = {row['id']: row for row in current_rank['boards']}
    changed_rank_ids = sorted(
        board_id for board_id in set(baseline_rank_by_id) | set(current_rank_by_id)
        if baseline_rank_by_id.get(board_id) != current_rank_by_id.get(board_id)
    )
    report['artifact_preservation'] = {
        'ranking_rows_exact': baseline_rank['boards'] == current_rank['boards'],
        'ranking_changed_rows': [
            {
                'id': board_id,
                'before': baseline_rank_by_id.get(board_id),
                'after': current_rank_by_id.get(board_id),
            }
            for board_id in changed_rank_ids
        ],
        'ranking_metadata_changed_keys': sorted(
            key for key in set(baseline_rank) | set(current_rank)
            if key != 'boards' and baseline_rank.get(key) != current_rank.get(key)
        ),
        'derived_boards_exact': baseline_existing == existing,
    }
    baseline_mobile = show_json(MOBILE_PATH)
    current_mobile = json.loads(MOBILE_PATH.read_text(encoding='utf-8'))
    report['artifact_preservation']['mobile_puzzles_exact'] = baseline_mobile['kg_puzzles'] == current_mobile['kg_puzzles']
    report['artifact_preservation']['mobile_expected_node_delta'] = [len(baseline_mobile['kg_nodes']), len(current_mobile['kg_nodes'])]
    report['artifact_preservation']['mobile_expected_edge_delta'] = [len(baseline_mobile['kg_edges']), len(current_mobile['kg_edges'])]
    report['artifact_preservation']['mobile_changed_top_keys'] = sorted(
        key for key in set(baseline_mobile) | set(current_mobile)
        if key != 'games' and baseline_mobile.get(key) != current_mobile.get(key)
    )

    old_nodes = set(baseline_svc.graph.nodes)
    report['graph_delta'] = {
        'nodes': [len(kg['kg_nodes']), len(prospective['kg_nodes'])],
        'edges': [len(kg['kg_edges']), len(prospective['kg_edges'])],
        'added_node_ids': sorted(set(prospective_svc.graph.nodes) - old_nodes),
        'removed_node_ids': sorted(old_nodes - set(prospective_svc.graph.nodes)),
        'nuca_outgoing': [
            {
                'dst': edge.dst_id,
                'relation': edge.relation,
                'label': edge.label_ro,
                'strength': edge.strength,
                'distractor': edge.is_distractor,
                'bidirectional': edge.bidirectional,
            }
            for edge in prospective_svc.graph.edges
            if edge.src_id == NUCA
        ],
        'nuca_predecessors': prospective_svc.predecessor_ids(NUCA),
    }

    output = SCRATCH / 'actual-all-six-impact.json'
    output.write_text(json.dumps(report, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
    print(json.dumps({
        'pack_counts': report['pack_counts'],
        'occurrences': len(report['occurrences']),
        'validation_changes': len(report['validation']),
        'contexto_changes': len(report['contexto']),
        'lant_changes': len(report['lant']),
        'alchimie_changes': len(report['alchimie']),
        'conexiuni_changes': len(report['conexiuni']),
        'derived': {k:v for k,v in report['derived'].items() if k not in {'removed','added','changed_common'}},
        'artifact_preservation': report['artifact_preservation'],
        'graph_delta': report['graph_delta'],
        'output': str(output),
        'sha256': sha256(output),
    }, ensure_ascii=False, indent=2))


if __name__ == '__main__':
    main()
