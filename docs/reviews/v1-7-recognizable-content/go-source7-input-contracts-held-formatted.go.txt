				// Reuse the public token rather than forge one from private session state.
				if row.Owner == "" && row.Projection == nil {
					challenge := request("POST", base+"/guess", map[string]any{"text": row.Text})
					repeatBody["confirm"] = challenge["resolved_token"]
				}
				repeated := request("POST", base+"/guess", repeatBody)
				if repeated["ok"] != true || repeated["won"] != false || repeated["attempts"] != float64(1) || !reflect.DeepEqual(repeated["guess"], guess) {
					t.Fatal("repeat changed the existing slot")
				}
				recover(repeated)
			} else {
				expected := make([]any, len(row.HTTPAdvisories))
				for i, label := range row.HTTPAdvisories {
					expected[i] = label
				}
				if first["ok"] != false || first["won"] != false || first["needs_confirmation"] == true || first["attempts"] != float64(0) || !reflect.DeepEqual(first["suggestions"], expected) {
					t.Fatalf("unknown input charged, acquired an owner, or changed normative advisories: %v", first)
				}
				repeated := request("POST", base+"/guess", map[string]any{"text": row.Text})
				if !reflect.DeepEqual(first, repeated) {
					t.Fatal("unknown repeat changed public state")
				}
				recover(repeated)
			}
		})
	}
	if !reflect.DeepEqual(counts, map[string]int{"exact": 9, "projection": 1, "fuzzy": 2, "unknown": 57}) {
		t.Fatalf("held69 class counts differ: %v", counts)
	}
}
