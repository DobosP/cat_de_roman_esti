def candidate_boards() -> list[dict]:
    return deepcopy(list(BOARDS))


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--repo", type=Path, required=True)
    args = parser.parse_args()
    repo = args.repo.resolve()
    sys.path.insert(0, str(repo))
    from cat_de_roman_esti.wordgames import quick_catalog as Q
    from cat_de_roman_esti.wordgames.derived_catalog import load_derived_catalog
    from cat_de_roman_esti.wordgames.service import get_service
    from scripts.quick_game_content_source_v92 import candidate_boards as prior_authored

    output = Path(__file__).resolve().parent
    frozen = repo / "cat_de_roman_esti/fixtures/derived_catalog_v38.json"
    supplement = repo / "cat_de_roman_esti/fixtures/quick_games_v92.json"
    pins = {str(p.relative_to(repo)): hashlib.sha256(p.read_bytes()).hexdigest()
            for p in [frozen, supplement, repo / "cat_de_roman_esti/fixtures/kg_sample.json",
                      repo / "scripts/quick_game_content_source_v92.py"]}
    baseline_pins = {
        relative: hashlib.sha256(subprocess.check_output(
            ["git", "show", f"c0ead5eb:{relative}"], cwd=repo)).hexdigest()
        for relative in pins
    }
    assert pins == baseline_pins, "protected baseline content changed"
    candidate = {"kind": "quick-content-candidates-v1", "boards": candidate_boards(),
                 "bindings": {**Q.bindings(live=False),
                              "editorial_source_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}}
    output.joinpath("candidates.json").write_bytes(Q.json_bytes(candidate))
    old_core = json.loads(frozen.read_bytes())["boards"]
    old_extra = json.loads(supplement.read_bytes())["boards"]
    old = old_core + old_extra
    assert len(old_core) == 336 and len(old_extra) == 45
    assert prior_authored() == json.loads(supplement.read_bytes())["authored"]
    svc = get_service()
    strengths = Q.graph_strengths(svc)
    # Replay the unchanged current structural gate over the full 45+6 supplement,
    # retaining the old 45 source rows exactly and using the original frozen base.
    combined = Q.rated_boards(prior_authored() + candidate_boards(), load_derived_catalog(), svc)
    old_pairs = {frozenset(p["members"]) for b in old if b["game"] == "perechi" for p in b["payload"]["pairs"]}
    seen_new_pairs = set()
    details = []
    for b, rated in zip(BOARDS, combined[45:], strict=True):
        ids = Q.visible_ids(b)
        same_game = [r for r in old if r["game"] == b["game"]]
        detail = {"id": b["id"], "labels": {i: svc.node(i).label_ro for i in ids},
                  "node_types": sorted({svc.node(i).node_type for i in ids}),
                  "minimum_salience": min(svc.node(i).salience for i in ids),
                  "old_maximum_visible_overlap": max(len(set(ids) & set(Q.visible_ids(r))) for r in same_game),
                  "standard_score": rated["standard_score"], "play_quality": rated["play_quality"],
                  "starter_eligible": rated["starter_eligible"]}
        if b["game"] == "intrusul":
            m, intruder = b["payload"]["members"], b["payload"]["intruder"]
            detail["inlier_strengths"] = [strengths.get(frozenset(p), 0) for p in itertools.combinations(m, 2)]
            detail["intruder_strengths"] = [strengths.get(frozenset([x, intruder]), 0) for x in m]
            detail["reused_old_trio"] = any(set(m) == set(r["payload"]["members"]) for r in same_game)
            assert not detail["reused_old_trio"]
        else:
            pairs = {frozenset(p["members"]) for p in b["payload"]["pairs"]}
            detail["intended_strengths"] = [strengths[frozenset(pair["members"])] for pair in b["payload"]["pairs"]]
            detail["maximum_cross_strength"] = max(strengths.get(frozenset(p), 0) for p in itertools.combinations(ids, 2) if frozenset(p) not in pairs)
            detail["reused_old_pairs"] = len(pairs & old_pairs)
            detail["reused_new_pairs"] = len(pairs & seen_new_pairs)
            assert not pairs & old_pairs and not pairs & seen_new_pairs
            seen_new_pairs.update(pairs)
        details.append(detail)
    # Mechanical matches under native strong edges: enumerate all 105 perfect
    # matchings, without claiming this proves natural-language uniqueness.
    def matching_count(ids):
        if not ids:
            return 1
        first, rest = ids[0], ids[1:]
        return sum(matching_count([x for x in rest if x != second])
                   for second in rest if strengths.get(frozenset([first, second]), 0) >= .60)
    for b, d in zip(BOARDS, details, strict=True):
        if b["game"] == "perechi":
            d["strong_graph_perfect_matchings"] = matching_count(Q.visible_ids(b))
            assert d["strong_graph_perfect_matchings"] == 1
    report = {"kind": "quick-content-author-probe-v1", "approval_status": "unapproved candidates; no independent reviews",
              "candidate_sha256": Q.digest(candidate), "baseline_commit": "c0ead5eb",
              "source_pins": pins, "baseline_pins_match": pins == baseline_pins, "preserved_core_boards": len(old_core),
              "preserved_existing_supplement_boards": len(old_extra),
              "candidate_counts": {g: sum(b["game"] == g for b in BOARDS) for g in ["intrusul", "perechi"]},
              "new_intended_pairs": len(seen_new_pairs), "boards": details}
    for relative, sha in pins.items():
        assert hashlib.sha256((repo / relative).read_bytes()).hexdigest() == sha
    output.joinpath("mechanical-checks.json").write_bytes(Q.json_bytes(report))
    print(json.dumps({"candidate_counts": report["candidate_counts"], "new_intended_pairs": len(seen_new_pairs), "existing_boards_preserved": 381, "candidate_sha256": report["candidate_sha256"]}))


if __name__ == "__main__":
    main()
