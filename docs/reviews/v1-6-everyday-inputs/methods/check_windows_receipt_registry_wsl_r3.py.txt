def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def write(path, value):
    blob = value if isinstance(value, bytes) else json.dumps(value).encode('utf-8') + b'\n'
    if len(blob) > MAX_FILE:
        raise ValueError('synthetic fixture file32KiB')
    with path.open('xb') as stream:
        stream.write(blob)


def snapshot(folder):
    values = {}
    for entry in folder.iterdir():
        metadata = entry.lstat()
        value = {'mode': metadata.st_mode, 'size': metadata.st_size}
        if stat.S_ISREG(metadata.st_mode):
            value['sha256'] = digest(entry)
        elif stat.S_ISLNK(metadata.st_mode):
            value['target'] = os.readlink(entry)
        values[entry.name] = value
    return values
