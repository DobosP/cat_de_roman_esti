import sys,json,random,copy
from pathlib import Path
ROOT=Path('/home/dobo/work/_worktrees/cat_de_roman_esti/feat__v92-alchimie-gui')
sys.path[:0]=[str(ROOT),str(ROOT/'scripts'),str(Path(__file__).parent)]
import critique_pack as C
from groups import GROUPS
from cat_de_roman_esti.wordgames.packs import validate_payload
pack,svc,strong,regions=C.load_all(ROOT/'cat_de_roman_esti/fixtures/games_pack.json',ROOT/'cat_de_roman_esti/fixtures/kg_sample.json')
out=Path(__file__).parent
bundles={}
def add(cat,game,row):
 bundles.setdefault(cat,{k:[] for k in ['nodes','edges','conexiuni','contexto','lant','alchimie']})[game].append(row)
cxgroups=[(0,1,2,10),(4,5,9,15),(3,11,30,26),(50,51,52,53),(55,56,57,58)]
relab={5:'Componente ale construcției unei case',9:'Ingrediente pentru îngroșarea sau închegarea preparatelor',56:'Mamifere cu copite',57:'Boabe și semințe comestibile',58:'Prăjituri coapte din aluat sau compoziție'}
for i,indices in enumerate(cxgroups):
 groups=[{'label':relab.get(j,GROUPS[j][0]),'tiles':[svc.resolve(t) for t in GROUPS[j][1].split('|')]} for j in indices]
 assert all(all(g['tiles']) for g in groups)
 add('limba' if i==3 else 'viata_de_roman','conexiuni',{'difficulty':'normal' if i in (1,3) else 'usor','groups':groups})
context=['Apă','Sare','Ou','Usturoi','Făină','Minge','Avion','Calculator','televizor','cheie','Ghiozdan','Muzeul Național al Satului „Dimitrie Gusti”','Florin Piersic','Amza Pellea','Dem Rădulescu','Arena Națională']
for label in context:
 n=svc.node(svc.resolve(label))
 assert n.id not in {r['target'] for r in pack['contexto']}
 add(n.category,'contexto',{'difficulty':'normal' if label in {'Muzeul Național al Satului „Dimitrie Gusti”','Dem Rădulescu'} else 'usor','target':n.id})
pairs=[('Zahăr','Cremșnit'),('Ion Creangă','Carte'),('Avion','București'),('Marea Neagră','Dunărea'),('Sibiu','Muzeu'),('Iași','Ion Creangă'),('Scrisoarea III','Floare albastră'),('Transilvania','Revoluția 1848'),('Carpați','Dunărea'),('Celentano','Mihai Bobonete'),('Românii au talent','Vocea României'),('România - Argentina 1994','Anghel Iordănescu'),('Helmut Duckadam','Emeric Ienei'),('David Popovici','Jocurile Olimpice Paris 2024'),('Ghiozdan','Ion Creangă'),('Electricitate','Dunărea')]
for a,b in pairs:
 start,target=svc.resolve(a),svc.resolve(b)
 assert start and target
 dist=svc.distance(start,target)
 add(svc.node(start).category,'lant',{'difficulty':'normal' if dist==3 else 'usor','start':start,'target':target})
private=copy.deepcopy(pack);records=[]
for cat,bundle in sorted(bundles.items()):
 path=out/'candidates'/cat/'candidates.json';path.parent.mkdir(parents=True,exist_ok=True)
 path.write_text(json.dumps(bundle,ensure_ascii=False,indent=2)+'\n')
 for game in ['conexiuni','contexto','lant']:
  for i,row in enumerate(bundle[game]):
   rec=copy.deepcopy(row)
   if game=='conexiuni':
    rec['groups']={f'g{j+1}':g['tiles'] for j,g in enumerate(row['groups'])};rec['group_labels']={f'g{j+1}':g['label'] for j,g in enumerate(row['groups'])};rec['order']=[n for g in row['groups'] for n in g['tiles']];random.Random(927+i).shuffle(rec['order'])
   if game=='lant':rec['optimal']=svc.distance(rec['start'],rec['target'])
   rec.update(id=f'preview_{game}_{cat}_{i:02d}',category=cat,source='ai',status='pending')
   errors=validate_payload(rec,game,svc)
   assert not errors,(rec['id'],errors)
   private[game].append(rec);records.append((game,rec))
ids={r['id'] for _,r in records}
items,packfindings,selected=C.run(private,svc,strong,regions,['conexiuni','contexto','lant'],{'pending'},ids)
dossiers={r['id']:C.build_dossier(r,game,svc,strong,findings,regions) for game,r,findings in selected}
(out/'preview-dossiers.json').write_text(json.dumps(dossiers,ensure_ascii=False,indent=2)+'\n')
(out/'preview-lints.json').write_text(json.dumps(items,ensure_ascii=False,indent=2)+'\n')
print(json.dumps({'counts':{g:sum(len(b[g]) for b in bundles.values()) for g in ['conexiuni','contexto','lant']},'findings':items},ensure_ascii=False,indent=2))
