def collect(value, path):
    if isinstance(value, dict):
        if value.get('game') in ('intrusul', 'perechi') and isinstance(value.get('payload'), dict):
            try:
                Q.visible_ids(value)
            except (KeyError, TypeError):
                pass
            else:
                archives.append((path, value))
        for part in value.values():
            collect(part, path)
    elif isinstance(value, list):
        for part in value:
            collect(part, path)

checks, new_pairs = [], set()
def count_matchings(vertices):
    if not vertices:
        return 1
    first, *rest = vertices
    return sum(count_matchings([x for x in rest if x != second]) for second in rest if weights.get(frozenset((first, second)), 0) >= .6)
for row in raw['boards']:
    game = row['game']
    visible = Q.visible_ids(row)
    same_game = [other for other in base_rows if other['game'] == game]
    prior_visible = {n for other in same_game for n in Q.visible_ids(other)}
    fresh = set(visible) - prior_visible
    overlaps = [len(set(visible) & set(Q.visible_ids(other))) for other in same_game]
    archive_same = [(path, other) for path, other in archives if other['game'] == game]
    archive_matches = [(path, other.get('id')) for path, other in archive_same if len(set(visible)&set(Q.visible_ids(other))) >= (3 if game=='intrusul' else 4)]
    assert not archive_matches
    result = {'id': row['id'], 'new_visible_words': [svc.label(n) for n in sorted(fresh)], 'max_installed_overlap': max(overlaps), 'historical_reskins': archive_matches}
    assert len(fresh) >= 3
    if game == 'intrusul':
        members, odd = row['payload']['members'], row['payload']['intruder']
        assert not any(set(members) == set(other['payload']['members']) for _, other in archive_same)
        result['inlier_edge_strengths'] = [weights.get(frozenset(pair), 0) for pair in itertools.combinations(members, 2)]
        result['outlier_edge_strengths'] = [weights.get(frozenset((odd, n)), 0) for n in members]
        assert min(result['inlier_edge_strengths']) >= .6 and max(result['outlier_edge_strengths']) == 0
    else:
        intended = {frozenset(pair['members']) for pair in row['payload']['pairs']}
        assert not intended & (used_pairs | new_pairs)
        assert not any(pair == frozenset(old_pair['members']) for pair in intended for _, other in archive_same for old_pair in other['payload']['pairs'])
        new_pairs.update(intended)
        result['intended_strengths'] = [weights.get(pair, 0) for pair in intended]
        result['maximum_cross_strength'] = max(weights.get(frozenset(pair), 0) for pair in itertools.combinations(visible, 2) if frozenset(pair) not in intended)
        result['native_perfect_matchings'] = count_matchings(visible)
        assert result['native_perfect_matchings'] == 1 and result['maximum_cross_strength'] == 0
    checks.append(result)
