def hash_file(path):
    h = hashlib.sha256()
    with Path(path).open('rb') as source:
        for chunk in iter(lambda: source.read(64 << 10), b''):
            h.update(chunk)
    return h.hexdigest()


def load_profile():
    if not sys.dont_write_bytecode or sys.flags.optimize != 0:
        raise RuntimeError('ordinary nonoptimized Python -B required before import')
    if hash_file(PROFILE) != PROFILE_SHA:
        raise RuntimeError('accepted profile source changed')
    loader = importlib.machinery.SourceFileLoader('runtime_mirror_exact_profile', str(PROFILE))
    spec = importlib.util.spec_from_loader(loader.name, loader)
    module = importlib.util.module_from_spec(spec)
    loader.exec_module(module)  # __main__ and the synthetic workload never run.
    if hash_file(PROFILE) != PROFILE_SHA or module.R != R or module.S != S or module.X_TASK != X:
        raise RuntimeError('profile source/roots changed during import')
    module.STARTED = STARTED
    module.CPU_STARTED = CPU_STARTED
    return module

@contextlib.contextmanager
def phase(name):
    global active_phase
    if len(phase_rows) >= 12 or active_phase is not None:
        raise RuntimeError('bounded nonnested phase records required')
    active_phase = {'name': name, 'started_elapsed_seconds': time.monotonic() - STARTED}
    try:
        yield
    except BaseException:
        active_phase['status'] = 'failed'
        raise
    else:
        active_phase['status'] = 'completed'
    finally:
        active_phase['duration_seconds'] = time.monotonic() - STARTED - active_phase['started_elapsed_seconds']
        phase_rows.append(active_phase)
        active_phase = None


def identity(value):
    return (value.st_dev, value.st_ino, value.st_size, value.st_mtime_ns, value.st_ctime_ns, value.st_nlink)


def directory_identity(value):
    if not stat.S_ISDIR(value.st_mode):
        raise RuntimeError('directory descriptor or named edge type differs')
    return value.st_dev, value.st_ino


class HeldTree:
    """Owned directory capabilities; no cached file size, hash or enumeration.

    Every original anchored parent->child edge is freshly checked at phase
    boundaries, including the canonical named root. File opens are relative to
    their held parent and use O_NOFOLLOW, with fresh leaf-entry/fstat identities.
    """
    def __init__(self, root):
        self.root = Path(root)
        self.records = {}
        self.directories = {}
        if not self.root.is_absolute() or '..' in self.root.parts:
            raise ValueError('literal absolute held root required')
        anchor = Path(self.root.anchor)
        fd = os.open(anchor, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW)
        COUNTS['directory_open'] += 1
        COUNTS['directory_fstat'] += 1
        try:
            wanted_anchor = directory_identity(os.fstat(fd))
        except BaseException:
            os.close(fd)
            raise
        self.records[anchor] = (fd, None, None, wanted_anchor)
        try:
            current = anchor
            for name in self.root.parts[1:]:
                guard()
                current = self._open_directory(current, name, make=False)
            self.directories[()] = current
            self.verify_edges()
        except BaseException:
            self.close()
            raise

    def _open_directory(self, parent, name, make):
        if not name or name in ('.', '..') or '/' in name or '\\' in name:
            raise ValueError('one literal directory component required')
        path = parent / name
        if path in self.records:
            return path
        if len(self.records) >= DIRECTORY_DESCRIPTOR_CAP:
            raise RuntimeError('held directory descriptor bound256')
        parent_fd = self.records[parent][0]
        try:
            COUNTS['directory_stat'] += 1
            named = os.stat(name, dir_fd=parent_fd, follow_symlinks=False)
        except FileNotFoundError:
            if not make:
                raise
            guard()
            os.mkdir(name, mode=0o700, dir_fd=parent_fd)
            guard()
            COUNTS['directory_stat'] += 1
            named = os.stat(name, dir_fd=parent_fd, follow_symlinks=False)
        wanted = directory_identity(named)
        fd = os.open(name, os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=parent_fd)
        COUNTS['directory_open'] += 1
        try:
            COUNTS['directory_fstat'] += 1
            if directory_identity(os.fstat(fd)) != wanted:
                raise RuntimeError('directory entry/descriptor identity changed while opening')
        except BaseException:
            os.close(fd)
            raise
        self.records[path] = (fd, parent, name, wanted)
        return path

    def directory_fd(self, parts=(), make=False):
        parts = tuple(parts)
        guard()
        if parts in self.directories:
            return self.records[self.directories[parts]][0]
        current = self.directories[()]
        traversed = ()
        for name in parts:
            guard()
            traversed += (name,)
            if traversed not in self.directories:
                current = self._open_directory(current, name, make)
                self.directories[traversed] = current
            else:
                current = self.directories[traversed]
        return self.records[current][0]

    def parent(self, relative, make=False):
        relative = Path(relative)
        if (relative.is_absolute() or '..' in relative.parts or not relative.parts or
                relative.as_posix() != '/'.join(relative.parts)):
            raise ValueError('closed literal relative file required')
        return self.directory_fd(relative.parts[:-1], make=make), relative.name

    def verify_edges(self):
        guard()
        for path, (fd, parent, name, wanted) in self.records.items():
            COUNTS['directory_fstat'] += 1
            if directory_identity(os.fstat(fd)) != wanted:
                raise RuntimeError('held directory descriptor identity drift')
            COUNTS['directory_stat'] += 1
            named = os.stat(path, follow_symlinks=False) if parent is None else os.stat(name, dir_fd=self.records[parent][0], follow_symlinks=False)
            if directory_identity(named) != wanted:
                raise RuntimeError('named directory ancestry edge changed: ' + str(path))
        guard()
        image = [(str(path), item[3]) for path, item in sorted(self.records.items())]
        return {'held_directory_count': len(image), 'edge_map_sha256': hashlib.sha256(json.dumps(image).encode()).hexdigest(),
                'fresh_all_named_edges_verified': True, 'named_root_identity': self.records[self.root][3]}

    def close(self):
        failures = []
        for fd, _, _, _ in reversed(list(self.records.values())):
            try:
                os.close(fd)
            except OSError as exc:
                failures.append(type(exc).__name__ + ': ' + str(exc))
        self.records.clear()
        self.directories.clear()
        if failures:
            raise RuntimeError('owned descriptor close failure: ' + '; '.join(failures))


def leaf(parent_fd, name):
    COUNTS['file_stat'] += 1
    value = os.stat(name, dir_fd=parent_fd, follow_symlinks=False)
    if not stat.S_ISREG(value.st_mode) or value.st_nlink != 1 or not 0 < value.st_size <= 16 << 20:
        raise RuntimeError('fresh single-link bounded regular leaf required: ' + name)
    return value


def fstat_file(fd):
    COUNTS['file_fstat'] += 1
    value = os.fstat(fd)
    if not stat.S_ISREG(value.st_mode) or value.st_nlink != 1:
        raise RuntimeError('opened file must remain single-link regular')
    return value


@contextlib.contextmanager
def read_leaf(tree, relative):
    guard()
    parent_fd, name = tree.parent(relative)
    initial = leaf(parent_fd, name)
    fd = os.open(name, os.O_RDONLY | os.O_NOFOLLOW, dir_fd=parent_fd)
    COUNTS['file_open'] += 1
    try:
        if identity(fstat_file(fd)) != identity(initial):
            raise RuntimeError('fresh leaf/opened file identity differs')
        yield fd, initial
        if identity(fstat_file(fd)) != identity(initial) or identity(leaf(parent_fd, name)) != identity(initial):
            raise RuntimeError('fresh leaf/file postimage identity differs')
    finally:
        os.close(fd)
    guard()


def hash_fd(fd, expected_bytes):
    os.lseek(fd, 0, os.SEEK_SET)
    value = hashlib.sha256()
    total = 0
    while True:
        guard()
        block = os.read(fd, CHUNK_BYTES)
        COUNTS['read_operations'] += 1
        COUNTS['bytes_read'] += len(block)
        guard()
        if not block:
            break
        total += len(block)
        if total > expected_bytes:
            raise RuntimeError('opened file exceeds fresh expected byte bound')
        value.update(block)
    if total != expected_bytes:
        raise RuntimeError('opened file byte count changed')
    return value.hexdigest()


def fresh_hash(tree, relative, expected_size=None):
    with read_leaf(tree, relative) as (fd, initial):
        if expected_size is not None and initial.st_size != expected_size:
            raise RuntimeError('fresh bound file size differs')
        return hash_fd(fd, initial.st_size), initial


def copy_one(row):
    relative = Path(row['path'])
    guard()
    destination_parent, name = mirror_tree.parent(relative)
    try:
        os.stat(name, dir_fd=destination_parent, follow_symlinks=False)
    except FileNotFoundError:
        pass
    else:
        raise FileExistsError('new mirror leaf already exists')
    with read_leaf(source_tree, relative) as (inp, initial):
        if initial.st_size != row['bytes'] or hash_fd(inp, row['bytes']) != row['sha256']:
            raise RuntimeError('canonical source preimage differs: ' + row['path'])
        if identity(fstat_file(inp)) != identity(initial):
            raise RuntimeError('source changed during pre-copy hash')
        out = os.open(name, os.O_WRONLY | os.O_CREAT | os.O_EXCL | os.O_NOFOLLOW, 0o600, dir_fd=destination_parent)
        COUNTS['file_open'] += 1
        value = hashlib.sha256()
        size = 0
        try:
            os.lseek(inp, 0, os.SEEK_SET)
            while True:
                guard()
                block = os.read(inp, CHUNK_BYTES)
                COUNTS['read_operations'] += 1
                COUNTS['bytes_read'] += len(block)
                if not block:
                    guard()
                    break
                size += len(block)
                if size > row['bytes']:
                    raise RuntimeError('streamed copy exceeds exact source bytes')
                value.update(block)
                remaining = memoryview(block)
                while remaining:
                    count = os.write(out, remaining)
                    COUNTS['write_operations'] += 1
                    if count <= 0:
                        raise RuntimeError('copy write made no progress')
                    COUNTS['bytes_written'] += count
                    remaining = remaining[count:]
                    guard()
                guard()
            os.fsync(out)
            COUNTS['file_fsync'] += 1
            destination = fstat_file(out)
        finally:
            os.close(out)
        if size != row['bytes'] or value.hexdigest() != row['sha256'] or identity(fstat_file(inp)) != identity(initial):
            raise RuntimeError('streamed copy/source identity differs')
        if hash_fd(inp, row['bytes']) != row['sha256']:
            raise RuntimeError('fresh canonical post-copy hash differs')
    actual, reread = fresh_hash(mirror_tree, relative, row['bytes'])
    if actual != row['sha256'] or identity(reread) != identity(destination):
        raise RuntimeError('fresh destination image/identity differs')
    os.fsync(destination_parent)
    COUNTS['directory_fsync'] += 1
    guard()
    return {'path': row['path'], 'bytes': size, 'sha256': value.hexdigest(),
            'source_dev': initial.st_dev, 'source_ino': initial.st_ino,
            'destination_dev': destination.st_dev, 'destination_ino': destination.st_ino,
            'source_unchanged': True, 'fresh_destination_verified': True}


def scan_entries(tree, parts):
    directory_fd = tree.directory_fd(parts)
    guard()
    scan_fd = os.open('.', os.O_RDONLY | os.O_DIRECTORY | os.O_NOFOLLOW, dir_fd=directory_fd)
    COUNTS['directory_open'] += 1
    try:
        COUNTS['directory_fstat'] += 2
        if directory_identity(os.fstat(scan_fd)) != directory_identity(os.fstat(directory_fd)):
            raise RuntimeError('fresh enumeration directory identity differs')
        COUNTS['scandir'] += 1
        with os.scandir(scan_fd) as entries:
            for entry in entries:
                guard()
                COUNTS['directory_stat'] += 1
                value = os.stat(entry.name, dir_fd=directory_fd, follow_symlinks=False)
                yield entry.name, value
    finally:
        os.close(scan_fd)
    guard()


def selected_runtime(relative):
    name = Path(relative).name
    return (not name.endswith('_test.go') and relative not in (
        'go-backend/internal/contentrails/sources/installed-authorities.json',
        'go-backend/internal/contentrails/installed_authorities_pin.go') and
        (name.endswith('.go') or name in ('go.mod', 'go.sum', 'rules_unicode15.json',
        'provenance.json', 'rules_provenance.json', 'bundled.json') or
        '/contentrails/sources/' in relative and name.endswith('.json')))


def runtime_inventory(tree, wanted):
    """Fresh complete directory enumeration; extra/missing/type/IO fail closed."""
    found = {}
    pending = [('go-backend',), ('shared-go', 'authcore')]
    while pending:
        parts = pending.pop()
        for name, value in scan_entries(tree, parts):
            relative_parts = parts + (name,)
            relative = '/'.join(relative_parts)
            if stat.S_ISDIR(value.st_mode):
                if name not in ('testdata', 'node_modules') and not name.startswith('.'):
                    pending.append(relative_parts)
            elif not stat.S_ISREG(value.st_mode):
                raise RuntimeError('runtime traversal symlink or nonregular entry')
            elif selected_runtime(relative):
                if relative in found or relative not in wanted:
                    raise RuntimeError('new/duplicate runtime selected file')
                found[relative] = fresh_hash(tree, relative)[0]
    if set(found) != set(wanted):
        raise RuntimeError('missing selected runtime file')
    return found


def exact_parity(entries, runtime):
    expected = {row['path']: row for row in entries}
    found = {}
    pending = [()]
    while pending:
        parts = pending.pop()
        for name, value in scan_entries(mirror_tree, parts):
            relative_parts = parts + (name,)
            relative = '/'.join(relative_parts)
            if stat.S_ISDIR(value.st_mode):
                pending.append(relative_parts)
            elif stat.S_ISREG(value.st_mode):
                if relative not in expected or relative in found:
                    raise RuntimeError('extra/duplicate mirror reader entry')
                row = expected[relative]
                found[relative] = fresh_hash(mirror_tree, relative, row['bytes'])[0]
                if found[relative] != row['sha256']:
                    raise RuntimeError('fresh mirror reader hash differs')
            else:
                raise RuntimeError('mirror nonregular or redirected entry')
    if set(found) != set(expected):
        raise RuntimeError('missing mirror reader files')
    for row in entries:
        if fresh_hash(source_tree, row['path'], row['bytes'])[0] != row['sha256']:
            raise RuntimeError('fresh canonical final parity differs')
    wanted = {row['path']: row['sha256'] for row in runtime}
    if len(wanted) != 109 or runtime_inventory(mirror_tree, wanted) != wanted or runtime_inventory(source_tree, wanted) != wanted:
        raise RuntimeError('fresh canonical/mirror runtime109 final enumeration differs')
    return {'reader_files': len(found), 'reader_bytes': sum(row['bytes'] for row in entries),
            'exact_runtime_files': 109, 'canonical_runtime_inventory_after_copy_exact': True,
            'all_source_destination_hashes_exact': True}

        report['status'] = 'completed_metadata_mirror_preparation'
        exit_code = 0
    except BaseException as exc:
        close_failures = []
        for tree in (source_tree, mirror_tree):
            if tree is not None:
                try:
                    tree.close()
                except BaseException as close_exc:
                    close_failures.append(type(close_exc).__name__ + ': ' + str(close_exc))
        source_tree = mirror_tree = None
        report.update(status='failed', failure=type(exc).__name__ + ': ' + str(exc),
                      bounded_phase_timings=phase_rows, adapter_operation_counts=COUNTS,
                      owned_descriptor_close_failures=close_failures,
                      all_owned_directory_descriptors_closed=not close_failures,
                      stream_chunk_bytes=CHUNK_BYTES,
                      completed_file_count=len(copied), Windows_metadata_cleanup='unconfirmed',
                      mirror=str(mirror) if mirror else None,
                      received_signal=p.received_signal if p is not None else None)
        if p is not None and p.runner is not None and not p.helper_active and p.helper_calls == p.helper_completed_calls:
            try:
                p.runner.validate_prior_windows_receipts(S / 'probe/windows-size-helper-r4')
                report['Windows_metadata_cleanup'] = 'confirmed_existing_terminal_registry'
            except BaseException as cleanup:
                report['cleanup_failure'] = type(cleanup).__name__ + ': ' + str(cleanup)
    try:
        if p is None or receipt_root is None:
            raise RuntimeError('no safe owned receipt root; capture stdout refusal')
        # Only reserved failure/publication work continues, never the failed copy.
        p.received_signal = None
        if p.runner is not None:
            p.runner.received_signal = None
        report['elapsed_seconds_before_publication'] = time.monotonic() - STARTED
        proof = p.exclusive_json(receipt_root / 'result.json', report, work=False)
        elapsed = time.monotonic() - STARTED
        if elapsed >= 50:
            raise TimeoutError('publication exceeded50seconds; receipt not accepted')
        signal.setitimer(signal.ITIMER_REAL, 0)
        print(json.dumps({'status': report['status'], 'exit_code': exit_code,
                          'total_elapsed_seconds': elapsed, 'result': proof}, sort_keys=True))
    except BaseException as exc:
        exit_code = 1
        signal.setitimer(signal.ITIMER_REAL, 0)
        print(json.dumps({'status': 'failed_publication', 'error': type(exc).__name__ + ': ' + str(exc),
                          'owned_receipt_root': str(receipt_root) if receipt_root else None,
                          'mirror': str(mirror) if mirror else None,
                          'total_elapsed_seconds': time.monotonic() - STARTED}, sort_keys=True))
    raise SystemExit(exit_code)
