def mutate(path, change):
    value = json.loads(path.read_bytes())
    change(value)
    path.write_bytes(B.json_bytes(value))

def _new_concept_review(reviewed_files, monkeypatch):
    """Exercise a new concept even during waves that only add recipes."""
    candidate = json.loads(reviewed_files[0].read_bytes())
    old = json.loads(B.BASELINE.read_bytes())
    concept_id = candidate["concepts"][0]["id"]
    old["concepts"] = [row for row in old["concepts"] if row["id"] != concept_id]
    baseline = reviewed_files[0].parent / "prior-without-concept.json"
    baseline.write_bytes(B.json_bytes(old))
    monkeypatch.setattr(B, "BASELINE", baseline)
    return candidate, concept_id


def _read_factual_concepts(reviewed_files, candidate):

def test_new_concept_cannot_forge_inherited_approval(reviewed_files, monkeypatch):

def test_reviewed_unchanged_concepts_can_inherit_without_new_source_claims(reviewed_files):
    old_ids = {c["id"] for c in json.loads(B.BASELINE.read_bytes())["concepts"]}
    def inherit(review):
        for row in review["concepts"]:
            if row["id"] in old_ids:
                row.update(inherited=True, sources=[])
    for path in reviewed_files[1:]:
        mutate(path, inherit)
    assert B.build_catalog(*reviewed_files)["concepts"]


# The two original positive inheritance contracts describe the V96 editorial
# baseline. Their original bodies stay exact; current-source controls are below.
_EDITORIAL_POSITIVE_CONTEXT_TESTS = frozenset({
    "test_editorial_descriptions_keep_original_fixes_and_provenance",
    "test_reviewed_unchanged_concepts_can_inherit_without_new_source_claims",
})
_EDITORIAL_HISTORICAL_KG_SHA = "d4774bb73d38500eada2d8f3c3a4b0829c660a2241d96f3e6826dd0ee862e109"
_EDITORIAL_BASELINE_SHA = "6de6b98871c43e539402fff9680474cc31afe22f5b4923c13dbafa9bd5465496"
_EDITORIAL_SOURCE_SHA = "66ade386c05e25f8b58990d127834c5803206e4565600256d55b09658bd23e2b"
_EDITORIAL_RUBRIC_SHA = "3fc2d6db8f8607d0bb70a9f7b4f329a42102b57ed2134e0f6e02ae5fb6e8e101"


def _editorial_context_read(path, expected_sha256, limit):
    path = Path(path)
    assert path.is_file() and not path.is_symlink()
    with path.open("rb") as stream:
        raw = stream.read(limit + 1)
    assert len(raw) <= limit
    assert hashlib.sha256(raw).hexdigest() == expected_sha256
    return raw


@contextmanager
def _historical_editorial_world_context(monkeypatch, temporary_root):
    """Use authenticated original KG bytes, then clear the scoped loader cache."""
    from cat_de_roman_esti.wordgames import service as service_loader

    assert Path(B.ROOT).resolve() == ROOT
    assert Path(B.BASELINE).resolve() == (
        ROOT / "docs/reviews/v96-words-and-input-clarity/integration/alchimie/candidate.json"
    )
    assert B.BASELINE_SHA == _EDITORIAL_BASELINE_SHA
    baseline = _editorial_context_read(B.BASELINE, _EDITORIAL_BASELINE_SHA, 2 << 20)
    assert json.loads(baseline)["bindings"]["kg_sha256"] == _EDITORIAL_HISTORICAL_KG_SHA
    assert Path(B.SOURCE.__file__).resolve() == ROOT / "scripts/alchimie_discovery_recipe_source.py"
    _editorial_context_read(B.SOURCE.__file__, _EDITORIAL_SOURCE_SHA, 1 << 20)
    compressed = _editorial_context_read(
        ROOT / "docs/reviews/v1-testing-release/content/kg-before-v1.json.gz",
        "b0cfac884d0d02d5218456fca77c0b91e1135238de2d5dde35f6636f78efcbf6",
        1 << 20,
    )
    assert len(compressed) == 431885
    with gzip.GzipFile(fileobj=io.BytesIO(compressed)) as stream:
        historical_kg = stream.read((4 << 20) + 1)
    assert len(historical_kg) == 3668778
    assert hashlib.sha256(historical_kg).hexdigest() == _EDITORIAL_HISTORICAL_KG_SHA
    rubric = _editorial_context_read(ROOT / "docs/CRITIQUE_RUBRIC.md", _EDITORIAL_RUBRIC_SHA, 32 << 10)
    assert not temporary_root.exists() and not temporary_root.is_symlink()
    temporary_root.mkdir(mode=0o700)
    for relative, raw in (
        ("cat_de_roman_esti/fixtures/kg_sample.json", historical_kg),
        ("docs/CRITIQUE_RUBRIC.md", rubric),
    ):
        destination = temporary_root / relative
        destination.parent.mkdir(parents=True, exist_ok=True)
        with destination.open("xb") as stream:
            stream.write(raw)
        assert destination.read_bytes() == raw
    fixture = temporary_root / "cat_de_roman_esti/fixtures/kg_sample.json"
    original_root = B.ROOT
    original_baseline = B.BASELINE
    original_source = B.SOURCE
    original_catalog = B.CATALOG
    service_loader.get_service.cache_clear()
    try:
        with monkeypatch.context() as scoped:
            scoped.setattr(B, "ROOT", temporary_root)
            scoped.setenv("CAT_KG_FIXTURE", str(fixture))
            try:
                yield temporary_root
            finally:
                service_loader.get_service.cache_clear()
    finally:
        service_loader.get_service.cache_clear()
        assert B.ROOT is original_root
        assert B.BASELINE is original_baseline
        assert B.SOURCE is original_source
        assert B.CATALOG is original_catalog

def test_current_alias_changed_concepts_cannot_claim_inherited_v96_approval(reviewed_files, tmp_path):
    assert Path(B.ROOT).resolve() == ROOT
    _editorial_context_read(
        ROOT / "cat_de_roman_esti/fixtures/kg_sample.json",
        "3412064a27c79e47b1b4205383e45c156316857bc8a6f6cb303bc65478501749",
        4 << 20,
    )
    candidate = json.loads(reviewed_files[0].read_bytes())
    previous = json.loads(_editorial_context_read(B.BASELINE, _EDITORIAL_BASELINE_SHA, 2 << 20))
    old = {row["id"]: row for row in previous["concepts"]}
    current = {row["id"]: row for row in candidate["concepts"]}
    changed = {identity for identity, row in old.items() if current[identity] != row}
    assert changed == {"n_v24_food_pantry_ulei", "n_v4gas_rosie"}
    original_review = json.loads(reviewed_files[1].read_bytes())
    recipe_ids = {row["id"] for row in candidate["recipes"]}
    for identity in sorted(changed):
        review = copy.deepcopy(original_review)
        row = next(row for row in review["concepts"] if row["id"] == identity)
        row.update(inherited=True, sources=[])
        path = tmp_path / ("invalid-inherited-" + identity + ".json")
        with path.open("xb") as stream:
            stream.write(B.json_bytes(review))
        with pytest.raises(ValueError, match="altered inherited concept"):
            B.read_review(
                path, "factual", B.file_sha(reviewed_files[0]), recipe_ids,
                candidate["concepts"],
            )


def test_historical_editorial_context_restores_root_environment_and_loader_after_error(monkeypatch, tmp_path):
    from cat_de_roman_esti.wordgames import service as service_loader

    original_root = B.ROOT
    original_baseline = B.BASELINE
    original_source = B.SOURCE
    original_catalog = B.CATALOG
    had_fixture = "CAT_KG_FIXTURE" in os.environ
    original_fixture = os.environ.get("CAT_KG_FIXTURE")
    with _historical_editorial_world_context(monkeypatch, tmp_path / "normal-exit") as historical:
        assert B.ROOT == historical and B.ROOT is not original_root
        assert os.environ["CAT_KG_FIXTURE"] == str(historical / "cat_de_roman_esti/fixtures/kg_sample.json")
        assert service_loader.get_service().resolve("untdelemn") is None
        assert service_loader.get_service().resolve("tomată") is None
        assert service_loader.get_service.cache_info().currsize == 1
    assert B.ROOT is original_root and B.BASELINE is original_baseline
    assert B.SOURCE is original_source and B.CATALOG is original_catalog
    assert ("CAT_KG_FIXTURE" in os.environ) == had_fixture
    assert os.environ.get("CAT_KG_FIXTURE") == original_fixture
    assert service_loader.get_service.cache_info().currsize == 0
    with pytest.raises(RuntimeError, match="owned context restoration control"):
        with _historical_editorial_world_context(monkeypatch, tmp_path / "exception-exit"):
            assert service_loader.get_service().resolve("untdelemn") is None
            raise RuntimeError("owned context restoration control")
    assert B.ROOT is original_root and B.BASELINE is original_baseline
    assert B.SOURCE is original_source and B.CATALOG is original_catalog
    assert ("CAT_KG_FIXTURE" in os.environ) == had_fixture
    assert os.environ.get("CAT_KG_FIXTURE") == original_fixture
    assert service_loader.get_service.cache_info().currsize == 0
