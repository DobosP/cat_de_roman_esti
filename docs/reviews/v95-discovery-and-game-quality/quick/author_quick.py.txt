archived = []

def visit(v, path):
    if isinstance(v, dict):
        if v.get('game') in ('intrusul', 'perechi') and isinstance(v.get('payload'), dict):
            try:
                Q.visible_ids(v)
            except (KeyError, TypeError):
                pass
            else:
                archived.append((path, v))
        for value in v.values():
            visit(value, path)
    elif isinstance(v, list):
        for value in v:
            visit(value, path)
for p in sorted((ROOT / 'docs/reviews').rglob('*.json')):
    try:
        before = len(archived)
        visit(json.loads(p.read_text()), str(p.relative_to(ROOT)))
        if len(archived) > before:
            archive_sources.append({'path': str(p.relative_to(ROOT)), 'sha256': sha(p)})
    except (ValueError, UnicodeError):
        pass

    comparisons = [(b._catalog_id, {'game': b.game, 'payload': b.payload}) for b in base if b.game == g]
    comparisons += [(path + '#' + a.get('id', ''), a) for path, a in archived if a['game'] == g]
    comparisons += [(a['id'], a) for a in selected if a['game'] == g]
    overlaps = [(name, len(visible & set(Q.visible_ids(a)))) for name, a in comparisons]
    maximum_overlap = max(count for _, count in overlaps)
    assert maximum_overlap < (3 if g == 'intrusul' else 4), (r['id'], overlaps)
    fresh = visible - used[g]
    assert len(fresh) >= 2, (r['id'], fresh)
    if g == 'intrusul':
        members, intruder = r['payload']['members'], r['payload']['intruder']
        assert all(frozenset(members) != frozenset(a['payload']['members']) for _, a in comparisons)
        mechanics = {'inside_strengths': [w.get(frozenset(p), 0) for p in itertools.combinations(members, 2)],
                     'intruder_strengths': [w.get(frozenset((m, intruder)), 0) for m in members]}
    else:
        intended = {frozenset(p['members']) for p in r['payload']['pairs']}
        for pair in intended:
            assert pair not in old_pairs | seen_pairs
            assert not [(name, a) for name, a in comparisons
                        if any(frozenset(p['members']) == pair for p in a['payload']['pairs'])]
            seen_pairs.add(pair)
        def matchings(remaining):
            if not remaining:
                return 1
            a = remaining[0]
            return sum(matchings([n for n in remaining if n not in (a, b)])
                       for b in remaining[1:] if w.get(frozenset((a, b)), 0) >= .6)
        mechanics = {'intended_strengths': [w.get(p, 0) for p in intended],
                     'maximum_cross_strength': max(w.get(frozenset(p), 0) for p in itertools.combinations(ids, 2) if frozenset(p) not in intended),
                     'native_perfect_matchings': matchings(ids)}
        assert mechanics['native_perfect_matchings'] == 1
    selected.append(r)

                    'archive_records_compared': sum(a['game'] == g for _, a in archived),
                    'mechanics': mechanics, 'rating': rating})
