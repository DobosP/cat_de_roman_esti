def canonical(v):
    return json.dumps(v, ensure_ascii=False, sort_keys=True, separators=(',', ':'),
                      allow_nan=False).encode()


def digest(v):
    return hashlib.sha256(canonical(v)).hexdigest()


def file_sha(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()


def read(p):
    return json.loads(Path(p).read_bytes())


def now():
    return dt.datetime.now(dt.timezone.utc)


def abort(*_):
    raise TimeoutError('shared admitted reference lane deadline reached')


def capture(fn, expected_error=None):
    # Only a source-specific exact refusal is an expected dependency observation.
    # Other errors escape to the structured failed-partition receipt.
    try:
        value = fn()
    except ValueError as exc:
        if expected_error is None or str(exc) != expected_error:
            raise
        return {'accepted': False, 'expected_dependency_refusal': True,
                'error_type': type(exc).__name__, 'message': str(exc)}
    assert expected_error is None, 'expected source dependency refusal did not occur'
    return {'accepted': True, 'value': value}


def plain(v):
    from collections.abc import Mapping
    if isinstance(v, Mapping):
        return {k: plain(x) for k, x in v.items()}
    if isinstance(v, (tuple, list)):
        return [plain(x) for x in v]
    return v

                rows = []
                for current in sorted(incident | {owner,nonneighbor}):
                    for spelling in (svc.label(owner),alias):
                        with patch.object(L,'get_service',return_value=svc):
                            resolved = L._resolve_neighbor(spelling,current,owner)
                        rows.append({'current':current,'text':spelling,'resolved':resolved,
                            'same_node':resolved == current,
                            'legal_link':resolved is not None and resolved != current and svc.link(current,resolved) is not None,
                            'caption':LR.caption(svc,current,resolved) if resolved else '',
                            'owner_to_current_allowed':svc.link(owner,current) is not None})
                return {'provenance':'constructed finite incident resolution, not ordinary selected play',
                        'owner':owner,'nonneighbor_control':nonneighbor,'rows':rows}
            raise AssertionError(args.stage)

        for index, case in enumerate(chosen,args.start):
            record = {'case':case,'index':index,'stage':args.stage,'status':'started',
                      'method_sha256':receipt['method_sha256'], 'admission_sha256':receipt['admission_sha256'],
                      'baseline_sha256':BASE_SHA,'candidate_sha256':CANDIDATE_SHA}
            case_path = out / f'{args.stage}-{index:04d}.json'
            assert not case_path.exists(), 'case overlap/overwrite forbidden'
            receipt['current_case'] = case
            checkpoint()
            for side,svc in enumerate(services):
                value = plain(profile(svc,side,case))
                record['before' if side == 0 else 'after'] = value
                # Save lossless completed baseline if candidate fails mid-case.
                if side == 0:
                    save(case_path,record)
            record.update(status='complete',before_sha256=digest(record['before']),after_sha256=digest(record['after']))
            record['changed'] = record['before'] != record['after']
            save(case_path,record,allow_replace=True)
            receipt['completed'].append({'id':case,'path':case_path.name,'sha256':file_sha(case_path),
                'before_sha256':record['before_sha256'],'after_sha256':record['after_sha256'],'changed':record['changed']})
            checkpoint()
        receipt['complete'] = True
        receipt['covered_count'] = len(receipt['completed'])
        receipt['changed_count'] = sum(x['changed'] for x in receipt['completed'])
        receipt['complete_stage'] = not receipt['missing_stage_ids']
        receipt['combined_before_sha256'] = digest([(x['id'],x['before_sha256']) for x in receipt['completed']])
        receipt['combined_after_sha256'] = digest([(x['id'],x['after_sha256']) for x in receipt['completed']])
    except BaseException as exc:
        receipt['failure'] = {'type':type(exc).__name__,'message':str(exc),
                              'traceback':traceback.format_exc(limit=8)}
    finally:
        signal.setitimer(signal.ITIMER_REAL,0)
        receipt['finished_at_utc'] = now().isoformat()
        receipt['installed_kg_exact'] = file_sha(BASE) == BASE_SHA
        receipt['candidate_kg_exact'] = file_sha(CANDIDATE) == CANDIDATE_SHA
        receipt['scratch_files_exact'] = all(file_sha(path) == expected
            for path, expected in admission['scratch_sha256'].items())
        receipt['source_files_exact'] = all(file_sha(ROOT / rel) == expected
                                            for rel, expected in admission['source_sha256'].items())
        checkpoint()
        print(json.dumps({'receipt':str(receipt_path),'complete':receipt['complete'],
                          'covered':len(receipt['completed']),'sha256':file_sha(receipt_path)}))
    return 0 if (receipt['complete'] and receipt['source_files_exact']
                 and receipt['scratch_files_exact']) else 1
