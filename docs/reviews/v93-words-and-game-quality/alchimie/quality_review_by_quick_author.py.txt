def walk(v):
    owned=set(v['world']['starter_ids']);crafted=set();tiers=set();witness={i:[] for i in owned};moves=[]
    while True:
        progress=False
        for u in v['unlocks']:
            if u['id'] not in tiers and len(crafted)>=u['after_discoveries']:
                tiers.add(u['id']);owned.update(u['concept_ids'])
                for i in u['concept_ids']:witness[i]=['supply:'+u['id']]
                progress=True
        for r in v['recipes']:
            if r['result'] not in owned and set(r['pair'])<=owned:
                owned.add(r['result']);crafted.add(r['result']);moves.append(r['id'])
                witness[r['result']]=list(dict.fromkeys(witness[r['pair'][0]]+witness[r['pair'][1]]+[r['id']]))
                progress=True
        if not progress:break
    assert owned=={x['id'] for x in v['concepts']}
    return owned,crafted,tiers,witness,moves
