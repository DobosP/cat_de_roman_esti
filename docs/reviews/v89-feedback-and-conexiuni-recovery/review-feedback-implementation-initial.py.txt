root=Path.cwd(); out=root/'docs/reviews/v89-feedback-and-conexiuni-recovery'
baseline='b614bfe735a27b9703025420daf603118f6a7f74'
pair=('n_v87_food_tort_diplomat','n_v86_food_frisca')
ast_checks={}
for file in ['cat_de_roman_esti/wordgames/contexto_feedback.py','cat_de_roman_esti/wordgames/contexto_projection.py']:
    old=ast.parse(subprocess.check_output(['git','show',f'{baseline}:{file}'],text=True))
    new=ast.parse((root/file).read_text())
    removed=0
    for node in new.body:
        if isinstance(node,ast.AnnAssign) and isinstance(node.target,ast.Name):
            if node.target.id=='EXACT_TARGET_FEEDBACK_PAIRS':
                arg=node.value.args[0]
                retained=[]
                for el in arg.elts:
                    if ast.literal_eval(el)==pair:
                        removed+=1
                    else:retained.append(el)
                arg.elts=retained
            if node.target.id=='PROJECTION_NEIGHBORHOODS':
                retained=[(k,v) for k,v in zip(node.value.keys,node.value.values) if ast.literal_eval(k)!='praf']
                removed+=len(node.value.keys)-len(retained)
                node.value.keys=[k for k,v in retained];node.value.values=[v for k,v in retained]
    assert removed==1
    assert ast.dump(old)==ast.dump(new),file
    ast_checks[file]={'whole_module_ast_exact_after_removing_one_approved_entry':True,'removed_entries':removed}
