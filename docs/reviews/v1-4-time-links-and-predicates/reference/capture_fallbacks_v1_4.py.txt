  return {k:('<session>' if k=='game_id' else norm(v)) for k,v in value.items()}
 if isinstance(value,list):return [norm(v) for v in value]
 if isinstance(value,str):

def traced(self,method,path,data='',content_type='application/octet-stream',secure=False,**kwargs):

 return response
