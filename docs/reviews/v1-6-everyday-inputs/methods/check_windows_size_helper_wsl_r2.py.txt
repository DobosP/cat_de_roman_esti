for name, arguments, result in (

    ('GetProcessTimes', (wintypes.HANDLE, ctypes.POINTER(wintypes.FILETIME), ctypes.POINTER(wintypes.FILETIME), ctypes.POINTER(wintypes.FILETIME), ctypes.POINTER(wintypes.FILETIME)), wintypes.BOOL),

    function.argtypes, function.restype = arguments, result

def checked(value):
    if not value:
        raise ctypes.WinError(ctypes.get_last_error())
    return value

def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def identity_quiescent(identity):
    if not isinstance(identity, dict) or type(identity.get('pid')) is not int or type(identity.get('creation_time_100ns')) is not int:
        raise ValueError('exact process identity required')
    handle = K.OpenProcess(0x100000 | 0x1000, False, identity['pid'])
    if not handle:
        if ctypes.get_last_error() == 87:
            return {'pid': identity['pid'], 'original_identity_quiescent': True, 'reason': 'PID absent'}
        raise ctypes.WinError(ctypes.get_last_error())
    try:
        c, e, k, u = (wintypes.FILETIME() for _ in range(4))
        checked(K.GetProcessTimes(handle, ctypes.byref(c), ctypes.byref(e), ctypes.byref(k), ctypes.byref(u)))
        actual = (int(c.dwHighDateTime) << 32) | int(c.dwLowDateTime)
        if actual != identity['creation_time_100ns']:
            return {'pid': identity['pid'], 'original_identity_quiescent': True, 'reason': 'PID reused; different creation identity left alone'}
        if K.WaitForSingleObject(handle, 0) != 0:
            raise RuntimeError('exact owned Windows process still running')
        return {'pid': identity['pid'], 'original_identity_quiescent': True, 'reason': 'matching process handle signalled exit'}
    finally:
        checked(K.CloseHandle(handle))

def junction(path, target):
    # An owned NTFS mount-point reparse entry needs no symlink privilege or shell command.
    path.mkdir()
    substitute = ('\\??\\' + str(target)).encode('utf-16-le')
    display = str(target).encode('utf-16-le')
    names = substitute + b'\0\0' + display + b'\0\0'
    data = struct.pack('<IHHHHHH', 0xA0000003, 8 + len(names), 0,
                       0, len(substitute), len(substitute) + 2, len(display)) + names
    handle = K.CreateFileW(str(path), 0x40000000, 7, None, 3, 0x02000000 | 0x00200000, None)
    if handle == ctypes.c_void_p(-1).value:
        raise ctypes.WinError(ctypes.get_last_error())
    try:
        returned = wintypes.DWORD()
        checked(K.DeviceIoControl(handle, 0x000900A4, data, len(data), None, 0, ctypes.byref(returned), None))
    finally:
        checked(K.CloseHandle(handle))
    if not path.stat(follow_symlinks=False).st_file_attributes & 0x400:
        raise RuntimeError('controlled junction was not a reparse entry')

def write_report(path, report):
    blob = json.dumps(report, indent=2).encode('utf-8') + b'\n'
    if len(blob) > 128 << 10:
        raise ValueError('fixture receipt bound')
    with path.open('wb') as stream:
        stream.write(blob)
        stream.flush()
        os.fsync(stream.fileno())

def main(directory, helper, helper_sha):
    if sys.platform != 'win32' or sys.version.split()[0] != '3.11.15' or Path(sys.executable).resolve() != PYTHON.resolve() or digest(PYTHON) != PYTHON_SHA:
        raise RuntimeError('exact native Windows interpreter required')
    base = (S / 'probe/windows-size-helper-fixtures').resolve()
    directory = Path(directory).resolve()
    if directory == base or os.path.commonpath((str(directory), str(base))) != str(base):
        raise ValueError('new owned fixture directory required')
    helper = Path(helper).resolve()
    if digest(helper) != helper_sha:
        raise RuntimeError('exact helper binding required')
    directory.mkdir(parents=True, exist_ok=False)
    tree = directory / 'tree'
    tree.mkdir()
    calls = []
    report = {'schema': 'v1-6-controlled-windows-size-helper-fixture-r2',

              'fixture_sha256': digest(__file__), 'calls': calls,
              'application_or_Go_executions': 0, 'foreign_processes_signalled': 0,
              'all_original_process_identities_quiescent': False}
    report_path = directory / 'check-result.json'
    write_report(report_path, report)
    def call(label, hang=False):
        if time.monotonic() - STARTED >= 25:
            raise TimeoutError('fixture total25s')
        receipt = S / 'probe/windows-size-helper-r3' / (uuid.uuid4().hex + '.json')
        argv = [str(PYTHON), '-B', str(helper), str(receipt)]
        if hang:
            argv.append('--fixture-hang')
        argv.append(tree.as_posix())
        began = time.monotonic()
        item = {'label': label, 'receipt_path': str(receipt), 'status': 'pending'}
        calls.append(item)
        write_report(report_path, report)
        process = subprocess.Popen(argv, stdin=subprocess.DEVNULL, stdout=subprocess.PIPE,
                                   stderr=subprocess.PIPE, creationflags=subprocess.CREATE_NO_WINDOW)
        c, e, k, u = (wintypes.FILETIME() for _ in range(4))
        checked(K.GetProcessTimes(wintypes.HANDLE(int(process._handle)), ctypes.byref(c),
                                  ctypes.byref(e), ctypes.byref(k), ctypes.byref(u)))
        item['native_call_launcher'] = {'pid': process.pid,
            'creation_time_100ns': (int(c.dwHighDateTime) << 32) | int(c.dwLowDateTime)}
        write_report(report_path, report)
        def preserve_streams(stdout, stderr, status):
            stdout = stdout or b''
            stderr = stderr or b''
            paths = {}
            for stream_name, data in (('stdout', stdout), ('stderr', stderr)):
                path = directory / (label + '.' + stream_name + '.bin')
                with path.open('xb') as stream:
                    stream.write(data[:1 << 20])
                    stream.flush()
                    os.fsync(stream.fileno())
                paths[stream_name] = {'path': str(path), 'original_bytes': len(data),
                    'saved_bytes': min(len(data), 1 << 20),
                    'sha256': hashlib.sha256(data).hexdigest(),
                    'truncated': len(data) > 1 << 20}
            item.update(status=status, exit_code=process.returncode,
                        parent_call_elapsed_seconds=time.monotonic() - began,
                        raw_streams=paths)
            write_report(report_path, report)
        try:
            stdout, stderr = process.communicate(timeout=5)
        except subprocess.TimeoutExpired as exc:
            # Never confuse terminating a venv redirector with terminating the helper.
            preserve_streams(exc.output, exc.stderr, 'cleanup-uncertain')
            raise RuntimeError('native helper call timeout; preserve pending identities and stop')
        elapsed = time.monotonic() - began
        # Save exact raw failure evidence and exit code before any shape/JSON assertion.
        preserve_streams(stdout, stderr, 'returned-unparsed')
        item['native_call_launcher_quiescence'] = identity_quiescent(item['native_call_launcher'])
        write_report(report_path, report)
        if len(stdout) > 1 << 20 or len(stderr) > 1 << 20 or elapsed > 4:
            raise RuntimeError('helper parent/output bound')

        saved = json.loads(receipt.read_bytes())

            raise RuntimeError('bindings drift during fixture call')

    try:

    except BaseException as exc:

    if report['elapsed_seconds'] > 25:
