results=[]

def no_secrets(state,board):
    assert not {board._catalog_id,board._source_id}&A.all_strings(state)

def hidden_labels(state,labels):
    assert not labels&A.all_strings(state)
    assert 'solution' not in state and 'score' not in state

for board in additions:
    if board._source_id not in new_source_ids:
        continue
    module=I if board.game=='intrusul' else P
    client=Client();url=f'/api/wordgames/{board.game}/games';gid=None
    with patch.object(module,'_catalog_or_503',return_value=combined):
        try:
            response=client.post(f'{url}?seed={seeds[board._catalog_id]}&category={board.category}')
            assert response.status_code==200
            state=response.json();gid=state['game_id'];endpoint=f'{url}/{gid}'
            assert module.store.get(gid).catalog_id==board._catalog_id
            no_secrets(state,board)
            start=client.get(endpoint).json();assert state==start
            def request(action,data=None,status=200):
                response=client.post(endpoint+'/'+action,data=data or {},content_type='application/json')
                assert response.status_code==status,(board._source_id,action,response.status_code,response.content)
                result=response.json();no_secrets(result,board);return result
            def unchanged_failure(action,data=None):
                before=client.get(endpoint).json()
                request(action,data,status=400)
                after=client.get(endpoint).json()
                assert before==after
            unchanged_failure('hint')
            events=[]
            if board.game=='intrusul':
                label=board.payload['group_label'];hidden_labels(start,{label})
                wrong=board.payload['members'][0]
                step=request('guess',{'id':wrong})
                assert not step['correct'] and step['mistakes']==1 and step['attempts']==1 and step['hint_available']
                hidden_labels(step,{label})
                earned=client.get(endpoint).json();assert earned['wrong_ids']==[wrong] and earned['mistakes']==1
                repeat=request('guess',{'id':wrong})
                assert repeat['already_tried'] and repeat['mistakes']==1 and repeat['attempts']==1
                assert client.get(endpoint).json()==earned
                hinted=request('hint');assert hinted['hints_used']==1 and 'clue' in hinted
                assert 'solution' not in hinted and 'score' not in hinted
                saved_hint=client.get(endpoint).json();assert saved_hint['clue']==hinted['clue']
                unchanged_failure('hint')
                repeat=request('guess',{'id':wrong});assert repeat['already_tried'] and repeat['mistakes']==1
                assert client.get(endpoint).json()==saved_hint
                won=request('guess',{'id':board.payload['intruder']})
                assert won['won'] and won['score']==650
                unchanged_failure('guess',{'id':board.payload['intruder']});unchanged_failure('hint')
                terminal=client.get(endpoint).json();assert terminal['won'] and terminal['score']==650
                events=['premature_hint_rejected_without_mutation','wrong_tap_costs_once','GET_restores_wrong_tap','repeat_wrong_free','one_hint_reveals_only_requested_group','GET_retains_hint','repeated_hint_rejected_without_mutation','repeat_wrong_after_hint_free','win_650','terminal_actions_rejected_without_mutation','terminal_GET_650']
            else:
                pairs=board.payload['pairs'];labels={p['group_label'] for p in pairs};hidden_labels(start,labels)
                first_wrong=[pairs[0]['members'][0],pairs[1]['members'][0]]
                step=request('match',{'ids':first_wrong});assert not step['correct'] and step['mistakes']==1 and step['actions']==1
                hidden_labels(step,labels);earned=client.get(endpoint).json();assert earned['mistakes']==1 and earned['actions']==1
                repeat=request('match',{'ids':list(reversed(first_wrong))});assert repeat['repeated'] and repeat['mistakes']==1 and repeat['actions']==1
                assert client.get(endpoint).json()==earned
                unchanged_failure('hint')
                second_wrong=[pairs[0]['members'][0],pairs[2]['members'][0]]
                second=request('match',{'ids':second_wrong});assert second['mistakes']==2 and second['actions']==2 and second['hint_available']
                hinted=request('hint');assert hinted['hints_used']==1
                hint_label=hinted['hint']['label'];assert hint_label in labels
                hidden_labels(hinted,labels-{hint_label})
                hint_state=client.get(endpoint).json();assert hint_state['hint']==hinted['hint'] and hint_state['mistakes']==2
                unchanged_failure('hint')
                solved=set()
                for index,pair in enumerate(pairs):
                    won=request('match',{'ids':list(pair['members'])});assert won['correct'] and won['solved_count']==index+1
                    solved.add(pair['group_label'])
                    if index<3:
                        hidden_labels(won,labels-solved-{hint_label})
                        before=client.get(endpoint).json();assert before['solved_count']==index+1
                        unchanged_failure('match',{'ids':list(reversed(pair['members']))})
                    else:

                unchanged_failure('match',{'ids':list(pairs[0]['members'])});unchanged_failure('hint')
                terminal=client.get(endpoint).json();assert terminal['won'] and terminal['score']==650
                events=['premature_hint_rejected_without_mutation','wrong_pair_costs_once','GET_restores_wrong_state','reversed_repeat_free','hint_after_one_mistake_rejected_without_mutation','second_wrong_unlocks_one_hint','only_hinted_pair_revealed','GET_retains_hint','repeated_hint_rejected_without_mutation','unsolved_labels_remain_hidden','GET_retains_partial_matches','repeat_solved_pair_rejected_without_mutation','win_650','terminal_actions_rejected_without_mutation','terminal_GET_650']
            no_secrets(terminal,board)

# Both additions must be naturally reachable through the real starter API.
starter_results=[]
for board in additions:
    if board._source_id not in new_source_ids:
        continue
    module=I if board.game=='intrusul' else P
    assert next(r['starter_eligible'] for r in catalog['boards'] if r['source_id']==board._source_id)
    client=Client();url=f'/api/wordgames/{board.game}/games'
    with patch.object(module,'_catalog_or_503',return_value=combined):
        for seed in range(2000):
            response=client.post(f'{url}?starter=1&seed={seed}&category={board.category}')
            assert response.status_code==200,response.content
            state=response.json();gid=state['game_id']
            if module.store.get(gid).catalog_id!=board._catalog_id:
                module.store.delete(gid)
                continue
            endpoint=f'{url}/{gid}'
            no_secrets(state,board)
            if board.game=='intrusul':
                response=client.post(endpoint+'/guess',{'id':board.payload['intruder']},content_type='application/json')
                assert response.status_code==200
                terminal=response.json()
            else:
                for pair in board.payload['pairs']:
                    response=client.post(endpoint+'/match',{'ids':pair['members']},content_type='application/json')
                    assert response.status_code==200
                terminal=response.json()
            assert terminal['won'] and terminal['score']==1000
            assert client.get(endpoint).json()['score']==1000
            no_secrets(terminal,board)
            starter_results.append({'id':board._catalog_id,'public_seed':seed,'starter':1,'category':board.category,'won':True,'score':1000})
            module.store.delete(gid)
            break
        else:
            raise AssertionError('new starter never selected')
