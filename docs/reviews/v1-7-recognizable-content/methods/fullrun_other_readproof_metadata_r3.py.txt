def inspect_source(name):
    if name in cache:
        return cache[name]
    filename = name
    text = (I / filename).read_text()
    tree = ast.parse(text)
    functions = {node.name: node for node in tree.body if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef))}
    fixtures = {name for name, node in functions.items() if any('pytest.fixture' in ast.unparse(d) for d in node.decorator_list)}
    autouse = {name for name in fixtures if any('autouse=True' in ast.unparse(d).replace(' ', '') for d in functions[name].decorator_list)}
    edges = {}
    for name, node in functions.items():
        edges[name] = {call.func.id for call in ast.walk(node) if isinstance(call, ast.Call) and isinstance(call.func, ast.Name) and call.func.id in functions}
        edges[name].update(arg.arg for arg in node.args.args if arg.arg in fixtures)
        if name.startswith('test'):
            edges[name].update(autouse)
    def callers(function):
        result = []
        for nodeid in cases['nodeids']:
            if nodeid.split('::')[0] != filename:
                continue
            pending = [nodeid.split('::')[-1].split('[')[0]]
            seen = set()
            while pending:
                item = pending.pop()
                if item not in seen:
                    seen.add(item)
                    pending.extend(edges.get(item, set()) - seen)
            if function in seen:
                result.append(nodeid)
        return result
    def container(line):
        nodes = [node for node in functions.values() if node.lineno <= line <= node.end_lineno]
        return min(nodes, key=lambda node: node.end_lineno - node.lineno).name if nodes else None
    readers = {name: [call.lineno for call in ast.walk(node) if isinstance(call, ast.Call) and isinstance(call.func, ast.Attribute) and call.func.attr in {'read_bytes', 'read_text', 'open'}] for name, node in functions.items()}
    readers = {name: lines for name, lines in readers.items() if lines}
    calls = []
    for call in ast.walk(tree):
        if not isinstance(call, ast.Call):
            continue
        leaf = call.func.attr if isinstance(call.func, ast.Attribute) else call.func.id if isinstance(call.func, ast.Name) else ''
        if leaf in {'read_bytes', 'read_text', 'open'} or (isinstance(call.func, ast.Name) and call.func.id in readers):
            calls.append(call)
    source_pins[str(I / filename)] = digest(I / filename)
    cache[filename] = (text, tree, callers, container, readers, calls)
    return cache[filename]

def proof(name, line, guard=None):
    text, tree, callers, container, readers, calls = inspect_source(name)
    statements = [node for node in ast.walk(tree) if isinstance(node, ast.stmt) and hasattr(node, 'lineno') and node.lineno <= line <= node.end_lineno]
    statement = min(statements, key=lambda node: node.end_lineno - node.lineno)
    targets = {node.id for node in getattr(statement, 'targets', []) if isinstance(node, ast.Name)}
    function = container(line)
    matches = [call for call in calls if call.lineno <= line <= call.end_lineno]
    if not matches and targets:
        matches = [call for call in calls if any(isinstance(node, ast.Name) and node.id in targets for node in ast.walk(call)) and (function is None or container(call.lineno) == function)]
    reads = []
    for call in matches:
        caller = container(call.lineno)
        if caller is None:
            continue
        helper = call.func.id if isinstance(call.func, ast.Name) else None
        reads.append({'line': call.lineno, 'expression': ast.get_source_segment(text, call), 'caller': caller, 'reader_helper_lines': readers.get(helper, []), 'collected_case_ids': callers(caller), 'guard': guard})
    return {'source': name, 'source_sha256': source_pins[str(I / name)], 'inventory_use_line': line, 'statement': ast.get_source_segment(text, statement), 'read_proofs': reads, 'classification': 'required immutable read proven' if reads else 'unresolved binding; this use grants no copy authority'}
