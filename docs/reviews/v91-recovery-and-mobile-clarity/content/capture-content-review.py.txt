def canonical(value):
    return json.dumps(value, ensure_ascii=False, sort_keys=True, separators=(",", ":")).encode()

def sha(value):
    return hashlib.sha256(canonical(value)).hexdigest()

def write(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")

def recipe(step):
    pair, outputs = step
    return {
        "pair": list(pair), "pair_labels": [svc.label(n) for n in pair],
        "outputs": list(outputs), "output_labels": [svc.label(n) for n in outputs],
        "actual_edges": [dataclasses.asdict(svc.link(parent, output))
                         for parent in pair for output in outputs],
    }

def book(rec):
    projection = a._build_recipe_projection(rec["seeds"], rec["target"], rec["category"])
    if projection is None:
        return None
    return {
        "par": projection.par,
        "recipes": [recipe(step) for step in projection.recipes.items()],
        "routes": [[recipe(step) for step in route] for route in projection.routes],
        "candidate_quality": projection.candidate_quality,
    }

def session(rec):
    projection = a._build_recipe_projection(rec["seeds"], rec["target"], rec["category"])
    assert projection and projection.par == rec["target_depth"]
    current = a.AlchimieSession(
        seeds=rec["seeds"], target=rec["target"], target_depth=projection.par,
        difficulty=rec["difficulty"], category=rec["category"], pack_id=rec["id"],
        recipes=projection.recipes, routes=projection.routes,
    )
    for node in current.seeds:
        current.add(node, None)
    return current

def profile(rec):
    current = session(rec)
    state = a._state_payload("content-review-local", current)
    owned = set(rec["seeds"])
    return {
        "record": rec,
        "record_sha256": sha(rec),
        "validation_errors": validate_payload(rec, "alchimie", svc),
        "initial_payload": state,
        "opening_pairs": [recipe((pair, outputs)) for pair, outputs in current.recipes.items()
                          if set(pair) <= owned and any(n not in owned for n in outputs)],
        "book": book(rec),
    }

def journey(rec, pairs):
    current = session(rec)
    game_id = a.store.create(current)
    client = Client()
    events = []
    try:
        response = client.get(f"/api/wordgames/alchimie/games/{game_id}")
        assert response.status_code == 200
        payload = response.json()
        payload["game_id"] = "ephemeral-content-review-session"
        assert payload["target"]["id"] is None
        assert "recipes" not in payload and "routes" not in payload
        events.append({"method": "GET", "status": response.status_code, "response": payload})
        for labels in pairs:
            pair = [svc.resolve(label) for label in labels]
            response = client.post(f"/api/wordgames/alchimie/games/{game_id}/combine",
                                   data={"a": pair[0], "b": pair[1]}, content_type="application/json")
            assert response.status_code == 200
            payload = response.json()
            payload["game_id"] = "ephemeral-content-review-session"
            events.append({"method": "POST combine", "pair": pair, "pair_labels": labels,
                           "status": response.status_code, "response": payload})
        assert payload["won"] and payload["moves"] == 2 and payload["score"] == 1000
        assert payload["target"]["id"] == rec["target"]
        response = client.get(f"/api/wordgames/alchimie/games/{game_id}")
        resumed = response.json()
        resumed["game_id"] = "ephemeral-content-review-session"
        assert resumed["won"] and resumed["score"] == 1000
        events.append({"method": "GET resume", "status": response.status_code, "response": resumed})
    finally:
        a.store.delete(game_id)
    return events
