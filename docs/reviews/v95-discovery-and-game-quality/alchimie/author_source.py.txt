def write_source():
    concepts = {label: {'id': 'alw_food_' + slug, 'description': description, 'sources': [S[key] for key in refs]}
                for label, slug, description, refs in concept_rows}
    raw = {'kind': 'v95-alchimie-author-proposals', 'concepts': concepts,
           'recipes': [{'id': r[0], 'pair_labels': list(r[1:3]), 'result_label': r[3],
                        'explanation': r[4], 'sources': [S[r[5]]]} for r in rows]}
    (OUT / 'raw/proposals.json').write_text(json.dumps(raw, ensure_ascii=False, sort_keys=True, indent=2) + '\n')
    path = ROOT / 'scripts/alchimie_discovery_recipe_source.py'
    source = path.read_text()
    marker = '\n# V95: familiar desserts with two approaches and reuse of prepared ingredients.\n'
    assert marker not in source, 'V95 has already been authored'
    source += marker
    source += 'WORLD_CONCEPTS.update(\n' + pprint.pformat(concepts, width=99, sort_dicts=False) + '\n)\n'
    source += 'RECIPES.extend(\n' + pprint.pformat([r[:5] for r in rows], width=99, sort_dicts=False) + '\n)\n'
    source += 'RECIPE_SOURCES.update(\n' + pprint.pformat({r[0]: [S[r[5]]] for r in rows}, width=99, sort_dicts=False) + '\n)\n'
    path.write_text(source)

if __name__ == '__main__':
    write_source()
