# A structural shortest-path DAG and par+2 corridor capture every eligible route node/edge.
# Menus are compared at the start and for one deterministic shortest prefix to each corridor node.
def bfs_chains(svc, start):
    parents={start:None}; queue=deque([start])
    while queue:
        cur=queue.popleft()
        for nxt in sorted(svc._adj[cur]):
            if nxt not in parents: parents[nxt]=cur; queue.append(nxt)
    chains={}
    for nid in parents:
        path=[]; cur=nid
        while cur is not None: path.append(cur); cur=parents[cur]
        chains[nid]=list(reversed(path))
    return chains

def lant_profile(svc, rec):
    start,target=rec['start'],rec['target']
    forward=svc.distances_from(start); reverse=svc.distances_to(target)
    optimal=reverse.get(start)
    strict={n for n,d in forward.items() if n in reverse and d+reverse[n]==optimal}
    corridor={n for n,d in forward.items() if n in reverse and d+reverse[n]<=optimal+2}
    edges={(a,b) for a in strict for b in svc._adj[a] if b in strict and forward[a]+1==forward[b]}
    chains=bfs_chains(svc,start)
    menus={}
    with patch.object(L,'get_service',new=lambda: svc):
        for n in sorted(corridor-{target}):
            session=L.LantSession(start,target,optimal,rec['difficulty'],chain=chains[n],category=rec['category'],pack_id=rec['id'])
            menus[n]={'chain':chains[n],'nodes':L._visible_choice_nodes(session),'choices':L._visible_choices(session)}
        profiles=L._compute_route_profiles(svc,start,target,optimal,slack=2)
    return {'optimal':optimal,'strict':strict,'corridor':corridor,'edges':edges,'menus':menus,
            'profiles':profiles,'errors':P.validate_payload(rec,'lant',svc)}
