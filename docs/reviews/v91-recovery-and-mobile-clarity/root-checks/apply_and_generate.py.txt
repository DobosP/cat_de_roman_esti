 log=OUT/(name+'.log');assert not log.exists(),name
 print('START',name,flush=True);start=time.monotonic()
 with log.open('w') as h:
  p=subprocess.Popen(command,cwd=ROOT,env=ENV,stdout=h,stderr=subprocess.STDOUT)
  while True:
   try:code=p.wait(timeout=30);break
   except subprocess.TimeoutExpired:print('RUNNING',name,round(time.monotonic()-start,1),flush=True)
 blob=log.read_bytes();record={'command':command,'cwd':str(ROOT),'exit_code':code,'elapsed_seconds':round(time.monotonic()-start,2),'log_sha256':hashlib.sha256(blob).hexdigest(),'tail':blob.decode(errors='replace').splitlines()[-6:]}
 log.with_suffix('.json').write_text(json.dumps(record,ensure_ascii=False,indent=2)+'\n');print('END',name,code,record['elapsed_seconds'],flush=True)
 if code:raise RuntimeError(name+' failed')
