def path(src, dst, weighted=False):
    frontier=[(0,src)]
    distances={src:0}
    prev={}
    while frontier:
        score,node=heapq.heappop(frontier)
        if score != distances[node]:
            continue
        if node==dst:
            steps=[]
            while node!=src:
                before=prev[node]
                edge=svc.link(before,node)
                steps.append({'from':svc.label(before),'to':svc.label(node),'forward_cost':svc._fwd_cost[before][node],'edge':dataclasses.asdict(edge)})
                node=before
            return {'cost':score,'steps':steps[::-1]}
        for nxt in sorted(svc.neighbor_ids(node)):
            cost=svc._fwd_cost[node][nxt] if weighted else 1
            new=score+cost
            if new < distances.get(nxt, math.inf):
                distances[nxt]=new
                prev[nxt]=node
                heapq.heappush(frontier,(new,nxt))
