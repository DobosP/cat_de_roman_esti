def closure(v):
    owned = set(v['world']['starter_ids'])
    crafted, tiers, witness = set(), set(), {n:[] for n in owned}
    while True:
        progressed = False
        for u in v['unlocks']:
            if u['id'] not in tiers and len(crafted) >= u['after_discoveries']:
                tiers.add(u['id']); owned.update(u['concept_ids']); progressed = True
                witness.update({n:['supply:'+u['id']] for n in u['concept_ids']})
        for r in v['recipes']:
            if r['result'] not in owned and set(r['pair']) <= owned:
                owned.add(r['result']); crafted.add(r['result']); progressed = True
                witness[r['result']] = list(dict.fromkeys(witness[r['pair'][0]]+witness[r['pair'][1]]+[r['id']]))
        if not progressed: break
    assert owned == {r['id'] for r in v['concepts']}
    return crafted, tiers, witness
