def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def info(pid):
    try:
        fields = (Path('/proc') / str(pid) / 'stat').read_text().rsplit(')', 1)[1].split()
        return {'pid': pid, 'state': fields[0], 'pgid': int(fields[2]),
                'sid': int(fields[3]), 'start_ticks': int(fields[19])}
    except (OSError, ValueError, IndexError):
        return None

def live(pid):
    value = info(pid)
    return value is not None and value['state'] not in ('Z', 'X')

def finalization_check():
    # A separately bounded Python-only fixture. The wrapper injects SIGINT only
    # at log hashing and returns the original digest; application data is absent.
    import importlib.util
    start = time.monotonic()
    deadline = start + 30
    directory = Path(sys.argv[1]).resolve()
    if not directory.is_relative_to(S / 'probe') or directory == S / 'probe':
        raise ValueError('owned probe directory required')
    directory.mkdir(mode=0o700)
    runner = S / 'probe/run_worktree_lane_wsl_r10.py'
    spec = importlib.util.spec_from_file_location('r4r2_guard_fixture', runner)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    # Controlled boundary values exercise the exact predicate used by the real
    # supervisor, without allocating1GiB or altering real resource thresholds.
    allowance = module.PUBLICATION_ALLOWANCE
    cap = module.GIB
    assert module.publication_guard(cap-allowance, 16*cap+allowance, 16*cap, 1024, False) is None
    assert module.publication_guard(cap-allowance+1, 16*cap+allowance, 16*cap, 1024, False)
    assert module.publication_guard(0, 16*cap+allowance-1, 16*cap, 1024, False)
    assert module.publication_guard(0, 17*cap, 16*cap, module.MAX_PUBLICATION_ENCODING+1, False)
    lane = 'r4-r2-finalize-' + hashlib.sha256(str(directory).encode()).hexdigest()[:16]
    manifest_path = directory / 'admission.json'
    marker = directory / 'late-signal.json'
    wrapper = directory / 'inject_late_signal.py'
    wrapper.write_text("""import importlib.util,json,os,pathlib,signal,sys
