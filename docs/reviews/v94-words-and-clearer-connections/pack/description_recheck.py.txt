"""Verify whether the suspect inherited target description reaches the player."""
import hashlib
import json
import os
from pathlib import Path
import sys

ROOT = Path('/home/dobo/work/_worktrees/cat_de_roman_esti/feat__v94-words-and-clearer-connections')
D = Path(__file__).resolve().parent
sys.path.insert(0, str(ROOT))
os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'cat_de_roman_esti.web.settings')
import django
django.setup()
from django.test import Client
from cat_de_roman_esti.wordgames import lant
from cat_de_roman_esti.wordgames.service import get_service

service = get_service()
start, target = 'n_v17geo_sfinxul_bucegi', 'n_v20geo_crucea_caraiman'
session = lant.LantSession(start=start, target=target, optimal=2, difficulty='usor', chain=[start])
sid = lant.store.create(session)
response = Client().get(f'/api/wordgames/lant/games/{sid}')
assert response.status_code == 200
state = response.json()
assert state['target']['description'] == service.description(target)
assert '28 de metri' in state['target']['description']
assert 'Șaua Caraiman' in state['target']['description']
ui_path = ROOT / 'frontend/src/screens/Lant.tsx'
ui_source = ui_path.read_text()
assert '<p className="muted lant-target-description">{state.target.description}</p>' in ui_source
