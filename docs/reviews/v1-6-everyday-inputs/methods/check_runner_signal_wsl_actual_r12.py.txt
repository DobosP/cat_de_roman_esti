def exact_bindings():

    # execution. This fixture's manifest is Python-only, never native admission.

    for path, wanted in ((runner, RUNNER_SHA), (MODE, MODE_SHA), (COPY_RESULT, COPY_RESULT_SHA)):
        value = path.lstat()
        if (not stat.S_ISREG(value.st_mode) or value.st_nlink != 1 or
                path.resolve(strict=True) != path or sha(path) != wanted):
            raise ValueError('exact unredirected fixture runner/mode/copy proof required')
    if not MIRROR.is_dir() or MIRROR.resolve(strict=True) != MIRROR:
        raise ValueError('closed direct mirror root differs')

            result.get('mirror') != str(MIRROR) or result.get('original_baseline_bytes') != 752431588 or

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def info(pid):
    try:
        fields = (Path('/proc') / str(pid) / 'stat').read_text().rsplit(')', 1)[1].split()
        return {'pid': pid, 'state': fields[0], 'pgid': int(fields[2]),
                'sid': int(fields[3]), 'start_ticks': int(fields[19])}
    except (OSError, ValueError, IndexError):
        return None

def live(pid):
    value = info(pid)
    return value is not None and value['state'] not in ('Z', 'X')
