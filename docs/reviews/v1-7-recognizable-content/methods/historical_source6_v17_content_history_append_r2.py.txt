def _v1_7_source6_blobs() -> dict[str, bytes]:
    """Read two finite exact original blobs from the independently reviewed archive."""
    import tarfile

    compressed = _V1_7_SOURCE6_ARCHIVE.read_bytes()
    assert len(compressed) == 5804665
    assert hashlib.sha256(compressed).hexdigest() == _V1_7_SOURCE6_ARCHIVE_SHA256
    result = {}
    with tarfile.open(fileobj=io.BytesIO(compressed), mode="r:gz") as archive:
        members = archive.getmembers()
        assert len(members) == 1114 and len({m.name for m in members}) == 1114
        assert all(member.isfile() for member in members)
        for name, (expected, size) in _V1_7_SOURCE6_MEMBER_PINS.items():
            member = archive.getmember(name)
            assert member.size == size and member.isfile()
            stream = archive.extractfile(member)
            assert stream is not None
            raw = stream.read(size + 1)
            assert len(raw) == size and hashlib.sha256(raw).hexdigest() == expected
            result[name] = raw
    return result


def historical_source6_v17_bytes() -> bytes:
    """Actual published Source6 fixture bytes; never consult the current candidate."""

def historical_source6_v17_hold_rows() -> list[dict]:
    """Frozen actual BEFORE outputs, independently reviewed; no current expectations."""
    raw = _v1_7_source6_blobs()["new-receipts/other69-reference-receipt.json"]
    receipt = json.loads(raw)
    assert receipt["complete"] and receipt["row_count"] == len(receipt["rows"]) == 69
    assert receipt["baseline_sha256"] == _V1_7_SOURCE6_MEMBER_PINS[
        "snapshots/physical-source6-kg.json"
    ][0]
    assert receipt["method_sha256"] == (
        "964ecbcb0e396e47f7033a3243cba80b6c470053da33c90d558232c1630832a0"
    )
    assert receipt["remaining69_sha256"] == _V1_7_SOURCE6_ORDERED69_SHA256
    assert not receipt["changed_lexical_rows"]
    assert all(row["before"] == row["after"] and not row["lexical_changed"]
               for row in receipt["rows"])
    surfaces = [row["surface"] for row in receipt["rows"]]
    encoded = json.dumps(surfaces, ensure_ascii=False, sort_keys=True,
                         separators=(",", ":")).encode()
    assert hashlib.sha256(encoded).hexdigest() == _V1_7_SOURCE6_ORDERED69_SHA256
    assert "leafă" not in surfaces
    return [{"surface": row["surface"], "expected_before": deepcopy(row["before"])}
            for row in receipt["rows"]]
