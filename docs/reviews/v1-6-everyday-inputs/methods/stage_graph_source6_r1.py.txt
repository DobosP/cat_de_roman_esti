"""Stage the accepted graph through its supported transaction and exact authored Source6.

Native ranking/derived builders run separately. No generated fixture is copied here.
"""
import hashlib
import json
import os
from pathlib import Path
import resource
import signal
import sys
import time

R = Path('/home/dobo/work/_worktrees/cat_de_roman_esti/codex__content-v1-6')
S = Path('/home/dobo/work/_temp/codex__content-v1-6')
D = R / 'docs/reviews/v1-6-everyday-inputs'
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
admission = json.loads(Path(sys.argv[1]).read_bytes())
assert sha(__file__) == admission['graph_staging_method_sha256']
assert all(sha(R / p) == h for p, h in admission['inputs'].items())
for name, key in [('graph-proposal-r1.json', 'graph_proposal_sha256'),
                  ('graph-factual-review-r1.json', 'graph_factual_sha256'),
                  ('remaining-native-controls-quality-r1.json', 'graph_quality_sha256'),
                  ('source6-authored-draft-r1.json', 'source6_sha256')]:
    assert sha(D / name) == admission[key]
proposal = json.loads((D / 'graph-proposal-r1.json').read_bytes())
assert all(sha(R / p) == h for p, h in proposal['source_bindings'].items())
assert all(sha(R / v['path']) == v['sha256'] for v in proposal['selected_pool_records'])
assert sha(R / 'scripts/everyday_inputs_v1_6_data.py') == admission['module_sha256']
source_target = R / 'go-backend/internal/contentrails/sources/authored-v6.json'
assert not source_target.exists(), 'Never replace an existing authored Source6'
source_bytes = (D / 'source6-authored-draft-r1.json').read_bytes()
source_temp = S / 'probe/authored-v6-staging-r1.json'
assert not source_temp.exists(), 'Preserve earlier staging attempts'
assert source_target.parent.stat().st_dev == source_temp.parent.stat().st_dev
sys.path[:0] = [str(R), str(R / 'scripts')]
resource.setrlimit(resource.RLIMIT_AS, (2 * 1024**3, 2 * 1024**3))
import apply_common_words_v24 as A
assert sha(A.__file__) == admission['inputs']['scripts/apply_common_words_v24.py']
expected_graph = admission['expected_after_graph']
assert {str(p.relative_to(R)) for p in A.TRANSACTION_FILES} == set(expected_graph)
snapshots = A._snapshot()
source_created = False
failure = None
started = time.monotonic()
def interrupted(signum, frame):
    raise RuntimeError('Outer resource/time guard interrupted staging; restore transaction')
signal.signal(signal.SIGTERM, interrupted)
try:
    A.apply(dry_run=False, module_name='everyday_inputs_v1_6_data')
    assert all(sha(R / p) == h for p, h in expected_graph.items()), 'Graph output differs from accepted generated bytes'
    with source_temp.open('xb') as out:
        out.write(source_bytes)
        out.flush()
        os.fsync(out.fileno())
    assert sha(source_temp) == admission['source6_sha256']
    assert not source_target.exists(), 'Concurrent Source6 creation; refuse replacement'
    os.replace(source_temp, source_target)
    source_created = True
    assert sha(source_target) == admission['source6_sha256']
except BaseException as exc:
    failure = {'type': type(exc).__name__, 'message': str(exc)}
    A._restore(snapshots)
    if source_created:
        assert sha(source_target) == admission['source6_sha256'], 'Do not remove unexpected authored bytes'
        source_target.unlink()
    raise
finally:
    result = {
        'schema': 'v1-6-actual-graph-source6-staging-r1',
        'method_sha256': sha(__file__),
        'elapsed_seconds': time.monotonic() - started,
        'max_RSS_KiB': resource.getrusage(resource.RUSAGE_SELF).ru_maxrss,
        'failure': failure,
        'graph_transaction_expected_bytes': {p: (R / p).exists() and sha(R / p) == h for p, h in expected_graph.items()},
        'authored_source6_present': source_target.exists(),
        'source6_sha256': sha(source_target) if source_target.exists() else None,
        'serving_export_or_current_authority_approved': False,
        'release_or_landing_approved': False,
    }
    if failure:
        result['graph_rollback_exact'] = all(p.read_bytes() == b for p, b in snapshots.items())
    (D / 'graph-source6-worktree-staging-r1-result.json').write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n')
