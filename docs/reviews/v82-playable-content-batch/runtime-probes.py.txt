def compact_json(value, level=0):
    """Keep scalar records on one line; nested collections stay navigable."""
    indent = '  ' * level
    if isinstance(value, dict) and any(isinstance(item, (dict, list)) for item in value.values()):
        entries = [json.dumps(key, ensure_ascii=False) + ': ' + compact_json(item, level + 1) for key, item in value.items()]
        return '{\n' + ',\n'.join('  ' + indent + entry for entry in entries) + '\n' + indent + '}'
    if isinstance(value, list) and any(isinstance(item, (dict, list)) for item in value):
        return '[\n' + ',\n'.join('  ' + indent + compact_json(item, level + 1) for item in value) + '\n' + indent + ']'
    return json.dumps(value, ensure_ascii=False, separators=(',', ':'))


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--root', type=Path, default=Path.cwd())
    parser.add_argument('--output', type=Path, required=True)
    parser.add_argument('--candidate-file', type=Path, default=Path('docs/reviews/v82-playable-content-batch/gastronomie/candidates.json'))
    parser.add_argument('--targets', nargs='+', help='Override the bounded sampled target list.')
    parser.add_argument('--words', nargs='+', help='Override the common guess list; exact answers and target-specific forms remain.')
    args = parser.parse_args()
    root = args.root.resolve()
    candidate = root / args.candidate_file
    batch = json.loads(candidate.read_text())['contexto']
    selected = {row['target']: row['difficulty'] for row in batch}
    source_paths = ['cat_de_roman_esti/wordgames/contexto.py', 'cat_de_roman_esti/wordgames/contexto_projection.py', 'cat_de_roman_esti/wordgames/contexto_feedback.py']
    source_bindings = {path: 'sha256:' + hashlib.sha256((root / path).read_bytes()).hexdigest() for path in source_paths}
    svc = get_service()
    client = Client()
    rows = []
    for target in (args.targets or DESSERT_TARGETS + SAVORY_TARGETS):
        difficulty = selected.get(target, 'normal')
        session = contexto._build_session(target, difficulty, None)
        row = {
            'target': target, 'label': svc.label(target), 'candidate': target in selected,
            'difficulty': difficulty, 'reachable': session.reachable,
            'responsive': sum(count for hops, count in session.dist_hist.items() if 1 <= hops <= 5),
            'incoming': [], 'outgoing': [], 'probes': [],
        }
        for direction, node_ids in [('incoming', svc.predecessor_ids(target)), ('outgoing', svc.neighbor_ids(target))]:
            for node_id in node_ids:
                edge = svc.link(node_id, target) if direction == 'incoming' else svc.link(target, node_id)
                row[direction].append({'id': node_id, 'label': svc.label(node_id), 'edge_id': edge.id, 'relation': edge.relation, 'strength': edge.strength})
        words = args.words or (DESSERT_WORDS if target in DESSERT_TARGETS else SAVORY_WORDS)
        for word in dict.fromkeys(words + [svc.label(target)] + EXTRA_WORDS.get(target, [])):
            game_id = contexto.store.create(contexto._build_session(target, difficulty, None))
            try:
                http_response = client.post(f'/api/wordgames/contexto/games/{game_id}/guess', {'text': word}, content_type='application/json')
                if http_response.status_code != 200:
                    raise RuntimeError(f'{target}/{word}: unexpected HTTP {http_response.status_code}')
                response = http_response.json()
                guess = response.get('guess', {})
                row['probes'].append({'text': word, 'accepted': bool(guess), 'guess_label': guess.get('label'), 'distance': guess.get('distance'), 'rank': guess.get('rank'), 'temperature': guess.get('temperature'), 'won': bool(response.get('won'))})
            finally:
                contexto.store.delete(game_id)
        rows.append(row)
    result = {
        'schema': 'v82-contexto-bounded-runtime-probes-v1',
        'candidate_sha256': 'sha256:' + hashlib.sha256(candidate.read_bytes()).hexdigest(),
        'kg_sha256': 'sha256:' + hashlib.sha256((root / 'cat_de_roman_esti/fixtures/kg_sample.json').read_bytes()).hexdigest(),
        'source_bindings': source_bindings,
        'scope': {'targets_sampled': len(rows), 'selected_candidates': sum(row['candidate'] for row in rows), 'fresh_session_guesses': sum(len(row['probes']) for row in rows), 'exhaustive': False, 'session_cleanup': 'Every created session deleted in finally; fixed target probes do not imply public eligibility.'},
        'targets': rows,
    }
    args.output.write_text(compact_json(result) + '\n')


if __name__ == '__main__':
    main()
