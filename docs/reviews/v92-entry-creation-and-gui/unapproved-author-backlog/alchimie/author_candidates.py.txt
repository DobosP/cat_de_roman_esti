def bytes_for(value: object) -> bytes:
    return (json.dumps(value, ensure_ascii=False, sort_keys=True, indent=2) + "\n").encode()


def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def candidate_witness(world: dict, proposal: dict) -> dict:
    """Produce the result first with this recipe, then finish the kitchen legally."""
    owned = set(world["world"]["starter_ids"])
    discovered: set[str] = set()
    unlocked: set[str] = set()
    trace = []
    first_index = None
    while True:
        gifts = [u for u in world["unlocks"] if u["id"] not in unlocked
                 and u["after_discoveries"] <= len(discovered)]
        for tier in gifts:
            owned.update(tier["concept_ids"])
            unlocked.add(tier["id"])
            trace.append({"unlock": tier["id"], "after_discoveries": len(discovered)})
        available = [r for r in world["recipes"]
                     if set(r["pair"]) <= owned and r["result"] not in owned]
        preferred = next((r for r in available if r["id"] == proposal["id"]), None)
        if first_index is None and preferred is None:
            available = [r for r in available if r["result"] != proposal["result"]]
        chosen = preferred or (available[0] if available else None)
        if chosen is None:
            break
        assert chosen["result"] not in owned
        assert set(chosen["pair"]) <= owned
        discovered.add(chosen["result"])
        owned.add(chosen["result"])
        trace.append({"recipe_id": chosen["id"], "pair": chosen["pair"],
                      "new_result": chosen["result"]})
        if chosen["id"] == proposal["id"]:
            first_index = len(trace) - 1
    assert first_index is not None, proposal["id"]
    assert len(discovered) == 117
    assert len(owned) == 221
    assert len(unlocked) == 12
    return {"recipe_id": proposal["id"], "first_use_trace_index": first_index,
            "craft_count": len(discovered), "owned_count": len(owned),
            "unlocked_tiers": len(unlocked), "trace": trace}
