def on_signal(number, frame):
    # Flag-only: Popen return-to-ownership bookkeeping cannot be interrupted.
    global received_signal
    received_signal=number
    if module is not None:
        module.received_signal=number

def exact_file(path, wanted, maximum=MAX_READ):
    path=Path(path)
    before=path.lstat()
    if (not stat.S_ISREG(before.st_mode) or before.st_nlink!=1 or
            path.resolve(strict=True)!=path or before.st_size>maximum):
        raise ValueError('exact single-link unredirected bounded regular file required: '+str(path))
    data=path.read_bytes()
    after=path.lstat()
    if (len(data)>maximum or
            (before.st_dev,before.st_ino,before.st_mode,before.st_size,before.st_mtime_ns,before.st_nlink)!=
            (after.st_dev,after.st_ino,after.st_mode,after.st_size,after.st_mtime_ns,after.st_nlink) or
            hashlib.sha256(data).hexdigest()!=wanted):
        raise ValueError('exact immutable bounded file differs: '+str(path))
    return data

def rss_bytes():
    for line in Path('/proc/self/status').read_text().splitlines():
        if line.startswith('VmRSS:'):
            return int(line.split()[1])*1024
    raise RuntimeError('own RSS unavailable')

def timestamp(value):
    result=datetime.datetime.fromisoformat(value)
    if result.tzinfo is None:
        raise ValueError('aware retained lane time required')
    return result
