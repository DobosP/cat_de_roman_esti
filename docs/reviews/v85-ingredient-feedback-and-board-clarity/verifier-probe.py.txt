def post(url,payload=None):
 r=client.post(url,payload or {},content_type='application/json');assert r.status_code==200,(url,r.status_code,r.content[:100]);return r.json()

def hidden(body,target):
 assert 'target' not in body and 'solution' not in body and 'anchor_id' not in str(body)
 assert target not in str(body) and svc.label(target).casefold() not in str(body).casefold()

 observed={}
 for word in words:
  gid=C.store.create(dataclasses.replace(template,guesses={},order=[]));url=f'/api/wordgames/contexto/games/{gid}'
  try:

  finally:C.store.delete(gid)

   hidden(client.get(url).json(),target)

  finally:C.store.delete(gid)
