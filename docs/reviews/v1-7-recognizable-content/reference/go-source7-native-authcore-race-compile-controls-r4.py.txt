def load_functions(path,names):
    nodes=[]
    for node in ast.parse(path.read_text()).body:
        if isinstance(node,ast.FunctionDef) and node.name in names:nodes.append(node)
        elif isinstance(node,ast.Assign) and isinstance(node.targets[0],ast.Name) and node.targets[0].id in ('REQUIRED_SPEC_KEYS','REQUIRED_STEP_KEYS','LATE_ADMISSION_KEYS'):nodes.append(node)
    env={}
    exec(compile(ast.Module(body=nodes,type_ignores=[]),str(path),'exec'),env)
    return env
WENV=load_functions(W,{'translated_windows_path','validate_contract','observe_linux_sample'})
LENV=load_functions(L,{'sample_supervisor','stop_sampler'})
class Event:
    def __init__(self,limit=3):self.set_flag=False;self.calls=0;self.limit=limit
    def is_set(self):return self.set_flag
    def set(self):self.set_flag=True
    def wait(self,seconds):
        self.calls+=1
        if self.calls>=self.limit:self.set()
class Thread:
    def __init__(self,alive=False):self.joined=False;self.alive=alive
    def join(self,timeout):self.joined=True
    def is_alive(self):return self.alive
class Cancel:
    def __init__(self):self.flag=False
    def exists(self):return self.flag
class Clock:
    def time(self):return 0

    def terminal(self,injection):
        tree=ast.parse(L.read_text())
        outer=next(n for n in tree.body if isinstance(n,ast.Try) and any(isinstance(x,ast.Expr) and isinstance(x.value,ast.Call) and isinstance(x.value.func,ast.Name) and x.value.func.id=='spec_pin' for x in n.body))
        start=next(i for i,n in enumerate(outer.body) if isinstance(n,ast.Expr) and isinstance(n.value,ast.Call) and isinstance(n.value.func,ast.Name) and n.value.func.id=='stop_sampler')
        events=[];cancel=Cancel()
        env={'stopping':False,'cancel':cancel,'spec':{'memory_floor':16,'deadline_epoch':1200,'rss_cap':10000},
             'time':type('Time',(),{'time':lambda self:0,'monotonic':lambda self:0})(),
             'mono_deadline':1200,'peak':10,'supervisor_peak':20,'step':{'name':'fixture'},
             'stop_sampler':lambda:events.append({'kind':'joined'})}
        def trip(where):
            if injection==where:env['stopping']=True
        def memory():trip('memory');return 100
        def group():trip('group');return [(1,1,1,10,'S')]
        def emit(value):events.append(value);trip('sample')
        env.update(memory=memory,group=group,emit=emit)
        code=compile(ast.Module(body=outer.body[start:],type_ignores=[]),str(L),'exec')
        return code,env,events
    def test_final_cancellation_never_completes(self):
        for where in ('memory','group','sample'):
            with self.subTest(where=where):
                code,env,events=self.terminal(where)
                with self.assertRaises(AssertionError):exec(code,env)
                self.assertFalse(any(x.get('kind')=='complete' for x in events))
    def test_terminal_joins_before_completion(self):
        code,env,events=self.terminal(None);exec(code,env)
        self.assertEqual(events[0]['kind'],'joined')
        self.assertEqual(events[-1]['kind'],'complete')
    def test_output_cap_predicate(self):

        class File:
            def __init__(self,size):self.size=size
            def stat(self):return type('Stat',(),{'st_size':self.size})()
        for out,err,expected in ((1<<20,1<<20,False),((1<<20)+1,0,True),(0,(1<<20)+1,True)):
            env={'pathlib':type('Paths',(),{'Path':lambda self,p:File(out)})(),'step':{'linux_output':'out','name':'x'},
                 'run':type('Run',(),{'__truediv__':lambda self,p:File(err)})()}
            self.assertEqual(eval(compile(ast.Expression(test),str(L),'eval'),env),expected)
