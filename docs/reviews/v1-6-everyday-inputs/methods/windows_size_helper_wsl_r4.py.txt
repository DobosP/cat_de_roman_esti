def checked(value):
    if not value:
        raise ctypes.WinError(ctypes.get_last_error())
    return value

def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def creation(handle):
    c, e, k, u = (wintypes.FILETIME() for _ in range(4))
    checked(K.GetProcessTimes(handle, ctypes.byref(c), ctypes.byref(e), ctypes.byref(k), ctypes.byref(u)))
    return (int(c.dwHighDateTime) << 32) | int(c.dwLowDateTime)

def memory_peak(handle):
    counters = MEMORY_COUNTERS()
    counters.cb = ctypes.sizeof(counters)
    checked(K.K32GetProcessMemoryInfo(handle, ctypes.byref(counters), counters.cb))
    peak = int(counters.PeakWorkingSetSize)
    if peak > MEMORY or int(counters.PeakPagefileUsage) > MEMORY:
        raise MemoryError('Windows helper process64MiB')
    return peak

def working_set_snapshot(handle):
    minimum, maximum, flags = ctypes.c_size_t(), ctypes.c_size_t(), wintypes.DWORD()
    checked(K.GetProcessWorkingSetSizeEx(handle, ctypes.byref(minimum), ctypes.byref(maximum), ctypes.byref(flags)))
    value = {'minimum_bytes': int(minimum.value), 'maximum_bytes': int(maximum.value),
             'enforcement_flags': int(flags.value)}
    if value['minimum_bytes'] != MIN_WORKING_SET or value['maximum_bytes'] != MEMORY or not flags.value & HARD_WORKING_SET_MAX or flags.value & 8:
        raise RuntimeError('exact1MiB/64MiB process hard working-set maximum not verified')
    return value

def working_set_enforce(handle, identity, scope, job=None, suspended=False):
    if time.monotonic() >= STARTED + WORK_CUTOFF:
        raise TimeoutError('hard working-set admission work cutoff')
    if type(identity.get('pid')) is not int or type(identity.get('creation_time_100ns')) is not int:
        raise ValueError('hard working-set exact identity shape')
    if checked(K.GetProcessId(handle)) != identity['pid'] or creation(handle) != identity['creation_time_100ns']:
        raise RuntimeError('hard working-set process identity differs before quota call')
    in_job = wintypes.BOOL()
    checked(K.IsProcessInJob(handle, job, ctypes.byref(in_job)))
    if job is not None:
        if scope != 'exact-owned-windows-job' or not in_job.value:
            raise RuntimeError('hard working-set process outside exact owned job')
    elif handle != K.GetCurrentProcess() or identity['pid'] != os.getpid() or scope not in ('parent-own-current-process', 'worker-own-current-process'):
        raise RuntimeError('hard working-set quota on foreign process refused')
    if scope == 'worker-own-current-process' and not in_job.value:
        raise RuntimeError('worker own quota requires existing Job membership')
    memory_peak(handle)
    checked(K.SetProcessWorkingSetSizeEx(handle, MIN_WORKING_SET, MEMORY, HARD_WORKING_SET_MAX))
    proof = dict(identity, **working_set_snapshot(handle), hard_max_verified=True,
                 identity_verified_before_quota=True, ownership_scope=scope,
                 job_membership_verified_before_quota=bool(in_job.value),
                 exact_owned_job_membership_verified_before_quota=job is not None,
                 enforced_while_suspended=bool(suspended), before_traversal=True,
                 quota_call_succeeded=True)
    memory_peak(handle)
    if time.monotonic() >= STARTED + WORK_CUTOFF:
        raise TimeoutError('hard working-set admission total work cutoff')
    return proof

def check_worker_self_proof(proof, identity):
    if not isinstance(proof, dict) or any(proof.get(key) != value for key, value in identity.items()):
        raise ValueError('worker self hard working-set identity proof')
    if proof.get('ownership_scope') != 'worker-own-current-process' or any(proof.get(key) is not True for key in
            ('hard_max_verified', 'identity_verified_before_quota', 'job_membership_verified_before_quota',
             'before_traversal', 'quota_call_succeeded')):
        raise ValueError('worker self hard working-set ownership proof')
    if proof.get('minimum_bytes') != MIN_WORKING_SET or proof.get('maximum_bytes') != MEMORY or type(proof.get('enforcement_flags')) is not int or not proof['enforcement_flags'] & 4 or proof['enforcement_flags'] & 8:
        raise ValueError('worker self hard working-set exact quota proof')

def canonical(path):
    value = Path(path)
    if not value.is_absolute() or '..' in value.parts:
        raise ValueError('absolute owned root required')
    resolved = value.resolve()
    if os.path.normcase(os.path.abspath(value)) != os.path.normcase(str(resolved)):
        raise ValueError('accounting root itself must not redirect')
    return resolved

def inside(path, base):
    return os.path.commonpath((os.path.normcase(str(path)), os.path.normcase(str(base)))) == os.path.normcase(str(base))

def roots_checked(values):
    if not 1 <= len(values) <= 32 or sum(map(len, values)) > 24000:
        raise ValueError('root argument bound')
    roots = [canonical(value) for value in values]
    bases = [R.resolve(), S.resolve()]
    if any(not any(inside(root, base) for base in bases) for root in roots):
        raise ValueError('root outside actual task/scratch directories')
    for i, root in enumerate(roots):
        if any(inside(root, prior) or inside(prior, root) for prior in roots[:i]):
            raise ValueError('overlapping accounting roots')
    return roots

def encoded(value, maximum=OUTPUT):
    blob = json.dumps(value, ensure_ascii=True, separators=(',', ':')).encode('utf-8') + b'\n'
    if len(blob) > maximum:
        raise ValueError('Windows helper output bound')
    return blob

def receipt_write(path, value, first=False):
    blob = encoded(value, 16 << 10)
    if first:
        with path.open('xb') as stream:
            stream.write(blob)
            stream.flush()
            os.fsync(stream.fileno())
    else:
        temporary = path.with_name(path.name + '.pending')
        with temporary.open('xb') as stream:
            stream.write(blob)
            stream.flush()
            os.fsync(stream.fileno())
        os.replace(temporary, path)

def job_empty(job):
    info = ACCOUNTING()
    checked(K.QueryInformationJobObject(job, 1, ctypes.byref(info), ctypes.sizeof(info), None))
    return info.ActiveProcesses == 0

def worker(arguments):
    source_sha, nonce, hang_text, *values = arguments
    if digest(__file__) != source_sha or digest(PYTHON) != PYTHON_SHA:
        raise RuntimeError('worker source/interpreter binding changed')
    self_identity = {'pid': os.getpid(), 'creation_time_100ns': creation(K.GetCurrentProcess())}
    self_proof = working_set_enforce(K.GetCurrentProcess(), self_identity, 'worker-own-current-process')
    roots = roots_checked(values)
    in_job = wintypes.BOOL()
    checked(K.IsProcessInJob(K.GetCurrentProcess(), None, ctypes.byref(in_job)))
    if not in_job.value:
        raise RuntimeError('worker not in a Windows job')
    ready = {'schema': 'v1-6-windows-size-worker-ready-r2', 'nonce': nonce,
             'source_sha256': source_sha, 'pid': os.getpid(),
             'creation_time_100ns': self_identity['creation_time_100ns'],
             'hard_working_set_self_proof': self_proof}
    ready_blob = encoded(ready, 4096)
    os.write(1, ready_blob)
    line = sys.stdin.buffer.readline(4097)
    if len(line) > 4096:
        raise ValueError('worker handshake bound')
    permission = json.loads(line)
    if permission.get('nonce') != nonce or permission.get('source_sha256') != source_sha or permission.get('worker_pid') != os.getpid():
        raise ValueError('worker start handshake mismatch')
    if permission.get('worker_creation_time_100ns') != ready['creation_time_100ns']:
        raise ValueError('worker creation identity mismatch')
    required_quota = {'minimum_bytes': MIN_WORKING_SET, 'maximum_bytes': MEMORY,
                      'hard_max_required': True}
    if permission.get('required_working_set_quota') != required_quota:
        raise ValueError('worker hard working-set quota handshake mismatch')
    working_set_snapshot(K.GetCurrentProcess())
    deadline = permission['work_deadline_monotonic']
    if not isinstance(deadline, (int, float)) or not STARTED < deadline <= STARTED + WORK_LIMIT:
        raise ValueError('worker time admission mismatch')
    if hang_text == '1':
        time.sleep(60)  # Controlled watchdog fixture; suspended launch guaranteed job ownership.
    pending, total, deferred, peak = list(roots), 0, [], 0
    def budget():
        nonlocal peak
        if time.monotonic() >= deadline:
            raise TimeoutError('Windows metadata work deadline')
        peak = max(peak, memory_peak(K.GetCurrentProcess()))
        working_set_snapshot(K.GetCurrentProcess())
    budget()
    while pending:
        directory = pending.pop()
        try:
            with os.scandir(directory) as entries:
                for entry in entries:
                    budget()
                    try:
                        metadata = entry.stat(follow_symlinks=False)
                        if metadata.st_file_attributes & 0x400:
                            deferred.append(entry.path)
                        elif entry.is_dir(follow_symlinks=False):
                            pending.append(entry.path)
                        elif entry.is_file():
                            total += metadata.st_size
                        else:
                            deferred.append(entry.path)
                    except FileNotFoundError:
                        pass
                    except OSError:
                        deferred.append(entry.path)
        except FileNotFoundError:
            pass
    budget()
    value = {'regular_bytes': total, 'deferred': deferred, 'peak_working_set_bytes': peak,
             'worker_elapsed_seconds': time.monotonic() - STARTED,
             'worker_working_set_at_completion': working_set_snapshot(K.GetCurrentProcess())}
    blob = encoded(value, OUTPUT - len(ready_blob))
    budget()
    if digest(__file__) != source_sha or digest(PYTHON) != PYTHON_SHA:
        raise RuntimeError('worker bindings changed during traversal')
    os.write(1, blob)

def supervisor(arguments):
    holder = {'job': None, 'launcher': None, 'worker': None}
    finished = threading.Event()
    def hard_guard():
        if finished.wait(max(0.0, HARD_PARENT_CUTOFF - (time.monotonic() - STARTED))):
            return
        if holder['job']:
            K.TerminateJobObject(holder['job'], 124)
        if holder['worker']:
            K.WaitForSingleObject(holder['worker'], 100)
        elif holder['launcher']:
            K.WaitForSingleObject(holder['launcher'], 100)
        os._exit(124)  # Non-inherited job handle closes; pending exact identities remain.
    threading.Thread(target=hard_guard, daemon=True).start()
    if len(arguments) < 2:
        raise ValueError('fresh owned receipt and accounting roots required')
    receipt = Path(arguments[0])
    hang = arguments[1:2] == ['--fixture-hang']
    values = arguments[2:] if hang else arguments[1:]
    roots = roots_checked(values)
    folder = S / 'probe/windows-size-helper-r4'
    folder.mkdir(parents=True, exist_ok=True)
    folder = canonical(folder)
    receipt = canonical(receipt)
    if receipt.parent != folder or not re.fullmatch(r'[0-9a-f]{32}\.json', receipt.name):
        raise ValueError('fresh per-call owned receipt location required')
    nonce = receipt.stem
    source_sha = digest(__file__)
    if Path(sys.executable).resolve() != PYTHON.resolve() or sys.version.split()[0] != '3.11.15' or digest(PYTHON) != PYTHON_SHA:
        raise RuntimeError('exact Windows interpreter required')
    if hang:
        fixture_base = S / 'probe/windows-size-helper-fixtures'
        if len(roots) != 1 or not inside(roots[0], fixture_base.resolve()):
            raise ValueError('hung-worker fixture root refused')
        authority = json.loads((roots[0].parent / 'allow-hung-worker.json').read_bytes())
        if authority != {'schema': 'v1-6-controlled-hung-worker', 'helper_sha256': source_sha}:
            raise ValueError('hung-worker fixture authorization refused')
    output_path = folder / (nonce + '.worker-output.json')
    ledger = {'schema': 'v1-6-windows-size-helper-owned-call-r2', 'nonce': nonce,
              'phase': 'created', 'roots': values, 'source_sha256': source_sha,
              'interpreter_sha256': PYTHON_SHA,
              'parent': {'pid': os.getpid(), 'creation_time_100ns': creation(K.GetCurrentProcess())},
              'launcher': None, 'worker': None, 'worker_exit_confirmed': False,
              'launcher_exit_confirmed': False, 'job_empty': False, 'job_terminated': False,
              'work_limit_seconds': WORK_LIMIT, 'parent_limit_seconds': PARENT_LIMIT,
              'job_memory_limit_bytes': MEMORY, 'process_memory_limit_bytes': MEMORY,
              'job_active_process_limit': 2, 'worker_output_path': str(output_path),
              'working_set_minimum_bytes': MIN_WORKING_SET, 'working_set_maximum_bytes': MEMORY,
              'hard_working_set_proofs': {}}
    receipt_write(receipt, ledger, first=True)
    job = None
    process, assigned, worker_exit, launcher_exit = PROCESS(), False, False, False
    input_read, input_write = wintypes.HANDLE(), wintypes.HANDLE()
    response, failure, attributes = None, None, None
    attributes_initialized = False
    ledger['job_setup'] = {'configured': False, 'pointer_bytes': ctypes.sizeof(ctypes.c_void_p),
                           'basic_limit_bytes': ctypes.sizeof(BASIC_LIMIT),
                           'extended_limit_bytes': ctypes.sizeof(EXTENDED_LIMIT),
                           'limit_flags': JOB_FLAGS}
    try:
        ledger['phase'] = 'parent-own-hard-working-set-admission'
        ledger['hard_working_set_proofs']['parent'] = working_set_enforce(K.GetCurrentProcess(), ledger['parent'], 'parent-own-current-process')
        receipt_write(receipt, ledger)
        job = checked(K.CreateJobObjectW(None, None))
        holder['job'] = job
        ledger['phase'] = 'job-created-before-configuration'
        receipt_write(receipt, ledger)
        limits = EXTENDED_LIMIT()
        limits.BasicLimitInformation.LimitFlags = JOB_FLAGS
        limits.BasicLimitInformation.ActiveProcessLimit = 2  # Redirector plus metadata worker.
        limits.ProcessMemoryLimit = limits.JobMemoryLimit = MEMORY
        checked(K.SetInformationJobObject(job, 9, ctypes.byref(limits), ctypes.sizeof(limits)))
        ledger['job_setup']['configured'] = True
        memory_peak(K.GetCurrentProcess())
        security = SECURITY(ctypes.sizeof(SECURITY), None, True)
        checked(K.CreatePipe(ctypes.byref(input_read), ctypes.byref(input_write), ctypes.byref(security), 4096))
        checked(K.SetHandleInformation(input_write, 1, 0))
        with output_path.open('xb') as output:
            output_handle = msvcrt.get_osfhandle(output.fileno())
            os.set_handle_inheritable(output_handle, True)
            startup = STARTUP_EX()
            startup.StartupInfo.cb = ctypes.sizeof(STARTUP_EX)
            startup.StartupInfo.dwFlags, startup.StartupInfo.wShowWindow = 0x101, 0
            startup.StartupInfo.hStdInput = input_read
            startup.StartupInfo.hStdOutput = startup.StartupInfo.hStdError = output_handle
            required = ctypes.c_size_t()
            K.InitializeProcThreadAttributeList(None, 1, 0, ctypes.byref(required))
            if required.value == 0 or required.value > 64 << 10:
                raise RuntimeError('Windows job-list attribute size unavailable')
            attributes = ctypes.create_string_buffer(required.value)
            checked(K.InitializeProcThreadAttributeList(attributes, 1, 0, ctypes.byref(required)))
            attributes_initialized = True
            jobs = (wintypes.HANDLE * 1)(job)
            # PROC_THREAD_ATTRIBUTE_JOB_LIST assigns atomically during creation.
            # A hard parent exit cannot leave an unassigned suspended launcher.
            checked(K.UpdateProcThreadAttribute(attributes, 0, 0x0002000D,
                                                ctypes.byref(jobs), ctypes.sizeof(jobs), None, None))
            startup.lpAttributeList = ctypes.cast(attributes, ctypes.c_void_p)
            argv = [str(PYTHON), '-B', str(Path(__file__).resolve()), '--worker',
                    source_sha, nonce, '1' if hang else '0', *map(str, roots)]
            command = ctypes.create_unicode_buffer(subprocess.list2cmdline(argv))
            checked(K.CreateProcessW(str(PYTHON), command, None, None, True,
                                     0x08000000 | 0x4 | 0x00080000, None, str(S),
                                     ctypes.cast(ctypes.byref(startup), ctypes.POINTER(STARTUP)), ctypes.byref(process)))
            assigned = True
            holder['launcher'] = process.hProcess
            ledger['launcher'] = {'pid': int(process.dwProcessId), 'creation_time_100ns': creation(process.hProcess)}
            in_job = wintypes.BOOL()
            checked(K.IsProcessInJob(process.hProcess, job, ctypes.byref(in_job)))
            if not in_job.value:
                raise RuntimeError('suspended launcher outside exact owned job')
            ledger['phase'] = 'owned-job-assigned-suspended'
            receipt_write(receipt, ledger)
            ledger['hard_working_set_proofs']['launcher'] = working_set_enforce(
                process.hProcess, ledger['launcher'], 'exact-owned-windows-job', job=job, suspended=True)
            ledger['phase'] = 'owned-suspended-launcher-hard-working-set-verified'
            receipt_write(receipt, ledger)
            if K.ResumeThread(process.hThread) == 0xFFFFFFFF:
                raise ctypes.WinError(ctypes.get_last_error())
            checked(K.CloseHandle(process.hThread))
            process.hThread = None
            checked(K.CloseHandle(input_read))
            input_read = wintypes.HANDLE()
            while True:
                if time.monotonic() >= STARTED + WORK_CUTOFF:
                    raise TimeoutError('worker ready watchdog')
                with output_path.open('rb') as stream:
                    ready_blob = stream.readline(4097)
                if ready_blob.endswith(b'\n'):
                    if len(ready_blob) > 4096:
                        raise ValueError('worker ready output bound')
                    ready = json.loads(ready_blob)
                    break
                if K.WaitForSingleObject(process.hProcess, 0) == WAIT_OBJECT_0:
                    raise RuntimeError('launcher exited before worker ready')
                time.sleep(0.005)
            if ready.get('schema') != 'v1-6-windows-size-worker-ready-r2' or ready.get('nonce') != nonce or ready.get('source_sha256') != source_sha:
                raise ValueError('actual worker ready binding')
            if type(ready.get('pid')) is not int or ready['pid'] <= 0 or type(ready.get('creation_time_100ns')) is not int:
                raise ValueError('actual worker identity shape')
            worker_handle = checked(K.OpenProcess(0x100000 | 0x1000 | 0x100, False, ready['pid']))  #SET_QUOTA
            holder['worker'] = worker_handle
            if creation(worker_handle) != ready['creation_time_100ns']:
                raise ValueError('actual worker creation identity changed')
            in_job = wintypes.BOOL()
            checked(K.IsProcessInJob(worker_handle, job, ctypes.byref(in_job)))
            if not in_job.value:
                raise RuntimeError('actual worker outside exact owned job')
            ledger['worker'] = {'pid': ready['pid'], 'creation_time_100ns': ready['creation_time_100ns']}
            ledger['phase'] = 'actual-worker-owned-not-started'
            receipt_write(receipt, ledger)
            self_proof = ready.get('hard_working_set_self_proof')
            check_worker_self_proof(self_proof, ledger['worker'])
            ledger['hard_working_set_proofs']['worker_self'] = self_proof
            ledger['hard_working_set_proofs']['worker'] = working_set_enforce(
                worker_handle, ledger['worker'], 'exact-owned-windows-job', job=job)
            ledger['launcher_working_set_before_traversal'] = working_set_snapshot(process.hProcess)
            ledger['phase'] = 'actual-worker-hard-working-set-verified-before-traversal'
            receipt_write(receipt, ledger)
            handshake = {'nonce': nonce, 'source_sha256': source_sha, 'worker_pid': ready['pid'],
                         'worker_creation_time_100ns': ready['creation_time_100ns'],
                         'work_deadline_monotonic': STARTED + WORK_CUTOFF,
                         'required_working_set_quota': {'minimum_bytes': MIN_WORKING_SET,
                              'maximum_bytes': MEMORY, 'hard_max_required': True}}
            blob = encoded(handshake, 4096)
            written = wintypes.DWORD()
            checked(K.WriteFile(input_write, blob, len(blob), ctypes.byref(written), None))
            if written.value != len(blob):
                raise RuntimeError('partial worker handshake')
            checked(K.CloseHandle(input_write))
            input_write = wintypes.HANDLE()
            remaining_ms = max(0, math.floor((STARTED + WORK_CUTOFF - time.monotonic()) * 1000))
            wait = K.WaitForSingleObject(worker_handle, remaining_ms)
            if wait == WAIT_TIMEOUT:
                raise TimeoutError('Windows owned-worker watchdog')
            if wait != WAIT_OBJECT_0:
                raise ctypes.WinError(ctypes.get_last_error())
            worker_exit = True
            ledger['worker_exit_elapsed_seconds'] = time.monotonic() - STARTED
            code = wintypes.DWORD()
            checked(K.GetExitCodeProcess(worker_handle, ctypes.byref(code)))
            ledger['worker_exit_code'] = int(code.value)
            if code.value != 0:
                raise RuntimeError('Windows metadata worker failed')
            if ledger['worker_exit_elapsed_seconds'] > WORK_LIMIT:
                raise TimeoutError('worker exit exceeds2s')
            if output_path.stat().st_size > OUTPUT:
                raise ValueError('worker output1MiB')
            lines = output_path.read_bytes().splitlines()
            if len(lines) != 2 or lines[0] + b'\n' != ready_blob:
                raise ValueError('worker result framing')
            response = json.loads(lines[1])
            if type(response.get('regular_bytes')) is not int or response['regular_bytes'] < 0 or not isinstance(response.get('deferred'), list):
                raise ValueError('worker result shape')
            for value in response['deferred']:
                if not isinstance(value, str) or not any(inside(Path(value), root) for root in roots):
                    raise ValueError('deferred path escaped roots')
            if not 0 < response.get('peak_working_set_bytes', 0) <= MEMORY:
                raise ValueError('worker working-set bound')
            completion_quota = response.get('worker_working_set_at_completion')
            if not isinstance(completion_quota, dict) or completion_quota.get('minimum_bytes') != MIN_WORKING_SET or completion_quota.get('maximum_bytes') != MEMORY or type(completion_quota.get('enforcement_flags')) is not int or not completion_quota['enforcement_flags'] & 4 or completion_quota['enforcement_flags'] & 8:
                raise ValueError('worker completion hard working-set proof')
    except BaseException as exc:
        failure = type(exc).__name__ + ': ' + str(exc)[:2048]
        ledger['failure_stage'] = ledger['phase']
        ledger['failure_winerror'] = getattr(exc, 'winerror', None)
    finally:
        if process.hProcess:
            remaining_ms = max(0, math.floor((STARTED + WORK_LIMIT - time.monotonic()) * 1000))
            launcher_exit = K.WaitForSingleObject(process.hProcess, remaining_ms if worker_exit else 0) == WAIT_OBJECT_0
            if not worker_exit or not launcher_exit or not job_empty(job):
                if assigned:
                    checked(K.TerminateJobObject(job, 124))
                    ledger['job_terminated'] = True
                else:
                    checked(K.TerminateProcess(process.hProcess, 124))
                    ledger['unassigned_owned_launcher_terminated'] = True
                if holder['worker']:
                    worker_exit = K.WaitForSingleObject(holder['worker'], 1000) == WAIT_OBJECT_0
                    ledger['worker_exit_elapsed_seconds'] = time.monotonic() - STARTED
                launcher_exit = K.WaitForSingleObject(process.hProcess, 1000) == WAIT_OBJECT_0
                if response is not None:
                    failure = failure or 'job required termination after normal worker result'
            ledger['worker_exit_confirmed'] = worker_exit
            ledger['launcher_exit_confirmed'] = launcher_exit
        else:
            # No successful CreateProcessW call: no launcher/worker identities are fabricated.
            ledger['launcher_not_created'] = ledger['worker_not_created'] = True
            ledger['cleanup_proof'] = 'no successful native process creation'
            worker_exit = launcher_exit = True
            ledger['worker_exit_confirmed'] = ledger['launcher_exit_confirmed'] = True
        ledger['job_empty'] = job_empty(job) if process.hProcess and job else True
        peak = EXTENDED_LIMIT()
        if job and process.hProcess:
            checked(K.QueryInformationJobObject(job, 9, ctypes.byref(peak), ctypes.sizeof(peak), None))
        elif not job:
            ledger['job_not_created'] = True
        else:
            ledger['peak_memory_proof'] = 'no successful native process creation'
        ledger['peak_job_memory_bytes'] = int(peak.PeakJobMemoryUsed)
        ledger['peak_worker_committed_memory_bytes'] = int(peak.PeakProcessMemoryUsed)
        if attributes_initialized:
            K.DeleteProcThreadAttributeList(attributes)
        for handle in (input_read, input_write, process.hThread, holder['worker'], process.hProcess):
            if handle:
                checked(K.CloseHandle(handle))
        if job:
            checked(K.CloseHandle(job))
        ledger['job_closed_confirmed'] = True
        holder.update(job=None, worker=None, launcher=None)
    memory_peak(K.GetCurrentProcess())
    if 'parent' in ledger['hard_working_set_proofs']:
        ledger['parent_working_set_at_completion'] = working_set_snapshot(K.GetCurrentProcess())
    if digest(__file__) != source_sha or digest(PYTHON) != PYTHON_SHA:
        failure = 'source/interpreter binding changed during call'
    if not worker_exit or not launcher_exit or not ledger['job_empty'] or ledger['peak_job_memory_bytes'] > MEMORY:
        failure = failure or 'Windows process exit/job/memory guard'
    if time.monotonic() - STARTED >= HARD_PARENT_CUTOFF:
        failure = failure or 'parent publication reserve exhausted'
    ledger.update(phase='failed' if failure else 'completed', failure=failure,
                  elapsed_seconds=time.monotonic() - STARTED)
    value = dict(ledger, schema='v1-6-windows-fresh-size-helper-r2',
                 receipt_schema='v1-6-windows-size-helper-owned-call-r2',
                 status=ledger['phase'], receipt_path=str(receipt), helper_sha256=source_sha,
                 python_version=sys.version.split()[0])
    if response is not None and failure is None:
        value.update(response)
    try:
        blob = encoded(value, 16 << 10)
    except ValueError:
        failure = 'final helper receipt16KiB'
        value = dict(ledger, schema='v1-6-windows-fresh-size-helper-r2',
                     receipt_schema='v1-6-windows-size-helper-owned-call-r2',
                     status='failed', phase='failed', failure=failure,
                     receipt_path=str(receipt), helper_sha256=source_sha,
                     python_version=sys.version.split()[0])
        blob = encoded(value, 16 << 10)
    if value['elapsed_seconds'] >= PARENT_LIMIT:
        raise TimeoutError('parent4s')
    receipt_write(receipt, value)
    # Exact same JSON values are committed to the final receipt and reported on stdout.
    os.write(1, blob)
    finished.set()
    return 1 if failure else 0

if __name__ == '__main__':
    if sys.platform != 'win32':
        raise SystemExit('Windows helper requires native Windows')
    if sys.argv[1:2] == ['--worker']:
        try:
            worker(sys.argv[2:])
        except BaseException as exc:
            os.write(1, encoded({'failure': type(exc).__name__, 'detail': str(exc)[:2048]}))
            raise SystemExit(1)
    else:
        raise SystemExit(supervisor(sys.argv[1:]))
