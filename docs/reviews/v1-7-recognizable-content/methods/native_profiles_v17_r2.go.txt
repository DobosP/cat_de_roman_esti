// JSONL records go directly to disk, so a killed run leaves its completed records.
// Cleanup adds a terminal receipt on ordinary failures/panics/Goexit. A hard kill
// cannot guarantee cleanup: absence of that terminal receipt means INCOMPLETE.
// No more than 8 MiB is emitted; overflow fails with explicit incomplete coverage.
type v17Proof struct {
    t *testing.T
    file *os.File
    bytes int
    counts map[string]int
    differences map[string]int
    complete bool
    stage string
}

func v17OpenProof(t *testing.T) *v17Proof {
    t.Helper()
    path := os.Getenv("V17_NATIVE_PROFILES_REPORT")
    if path == "" || !filepath.IsAbs(path) { t.Fatal("V17_NATIVE_PROFILES_REPORT must be a fresh absolute scratch JSONL path") }
    f, err := os.OpenFile(path, os.O_WRONLY|os.O_CREATE|os.O_EXCL, 0600)
    if err != nil { t.Fatal(err) }
    p := &v17Proof{t:t, file:f, counts:map[string]int{}, differences:map[string]int{}, stage:"starting"}
    t.Cleanup(func() {
        // Reserve 64 KiB for this final record even if a delta exhausted the budget.
        row := map[string]any{"kind":"terminal", "at":time.Now().UTC().Format(time.RFC3339Nano), "complete":p.complete, "test_failed":t.Failed(), "last_stage":p.stage, "counts":p.counts, "differences":p.differences, "bytes_before_terminal":p.bytes, "qualification":"research_only; no sealed export, current authority, target approval, human or HTTP evidence"}
        raw, err := json.Marshal(row)
        if err == nil { _, err = f.Write(append(raw, '\n')) }
        if err != nil { t.Errorf("final proof receipt: %v", err) }
        if err := f.Sync(); err != nil { t.Errorf("proof sync: %v", err) }
        if err := f.Close(); err != nil { t.Errorf("proof close: %v", err) }
    })
    return p
}
func (p *v17Proof) emit(row any) {
    p.t.Helper()
    raw, err := json.Marshal(row)
    if err != nil { p.t.Fatalf("proof encoding at %s: %v", p.stage, err) }
    if p.bytes+len(raw)+1 > (8<<20)-(64<<10) { p.t.Fatal("proof byte ceiling reached; coverage incomplete, inspect retained JSONL") }
    n, err := p.file.Write(append(raw,'\n')); p.bytes += n
    if err != nil { p.t.Fatalf("proof write: %v",err) }
}
func (p *v17Proof) same(scope, key string, before, after any) bool {
    p.counts[scope]++
    if reflect.DeepEqual(before,after) { return true }
    p.differences[scope]++
    p.emit(map[string]any{"kind":"unexpected_delta", "scope":scope,"key":key,"before":before,"after":after})
    return false
}
func (p *v17Proof) need(ok bool, message string) {
    p.t.Helper()
    if !ok { p.emit(map[string]any{"kind":"failed_precondition","stage":p.stage,"message":message}); p.t.Fatal(message) }
}
func v17Digest(v any) string {
    raw, err := json.Marshal(v)
    if err != nil { panic(err) }
    digest := sha256.Sum256(raw)
    return hex.EncodeToString(digest[:])
}
func v17HashRow(h hash.Hash, row any) {
    raw, err := json.Marshal(row)
    if err != nil { panic(err) }
    h.Write(raw); h.Write([]byte{'\n'})
}
func v17Bits(values []float64) []uint64 {
    out := make([]uint64,len(values)); for i,v := range values { out[i]=math.Float64bits(v) }; return out
}
func v17ProfileView(v *profile) any {
    buckets := map[int][]uint64{}
    for k,row := range v.Buckets { buckets[k]=v17Bits(row) }
    return struct { Dist []int32; WeightedBits []uint64; Buckets map[int][]uint64; Closer map[int]int; Reachable int }{v.Dist,v17Bits(v.Weighted),buckets,v.Closer,v.Reachable}
}
// The production FIFO limit remains 256. Only this serial test's own service
// pointer is detached from its Content shared cache; each fresh private cache
// retains one target. No live process, production limit, or cache code is changed.
func v17FreshProfile(s *Service,target string) *profile {
    s.profiles=&profileCache{values:map[string]*profile{}}
    return s.profile(target)
}
type v17FeedbackRow struct {
    ID,Label,Submitted,Anchor,Temperature,WinnerID string
    Reachable bool
    Distance,Rank,Closeness,Penalty int
    AnchorDistance int32
    AnchorWeightedBits uint64
}
func v17Feedback(s *Service,v *game,id string,term *projection) v17FeedbackRow {
    submitted,label,publicID,penalty := id,s.g.Label(id),id,0
    if term!=nil { submitted=s.projectionAnchor(term,v.Target); label=term.Label; publicID=term.PublicID; penalty=term.RankPenalty }
    sc:=s.score(v,submitted,term!=nil,penalty)
    d:=999; if sc.Reachable { d=sc.Distance }
    row:=v17FeedbackRow{ID:publicID,Label:label,Submitted:submitted,Anchor:sc.Anchor,Temperature:temperature(v,sc),Reachable:sc.Reachable,Distance:d,Rank:sc.Rank,Closeness:closeness(v,sc),Penalty:penalty,AnchorDistance:-1,AnchorWeightedBits:math.Float64bits(math.Inf(1))}
    if ix,ok:=s.g.Index(sc.Anchor); ok { row.AnchorDistance=v.Profile.Dist[ix]; row.AnchorWeightedBits=math.Float64bits(v.Profile.Weighted[ix]) }
    // Matches GuessInput's winner test; a proxy/exact-pair anchor is not a win.
    if term==nil && submitted==v.Target { row.WinnerID=v.Target }
    return row
}
type v17Resolution struct {
    Normalized,Owner,Fuzzy string
    Blocked bool
    Suggestions []string
    Projection *projection
    ProjectionSuggestions []projection
}
func v17Resolve(s *Service,text string,full bool) v17Resolution {
    key:=s.g.Normalize(text)
    row:=v17Resolution{Normalized:key,Owner:s.g.Resolve(text),Fuzzy:s.g.ResolveFuzzy(text),Blocked:s.g.ReviewedUnresolved(text),Projection:s.projections[key]}
    if full {
        row.Suggestions=s.g.Suggest(text,3)
        row.ProjectionSuggestions=[]projection{}
        for _,term:=range s.suggestProjection(text) { row.ProjectionSuggestions=append(row.ProjectionSuggestions,*term) }
    }
    return row
}
func v17SanitizeBody(body map[string]any) map[string]any {
    if body==nil { return nil }
    out:=map[string]any{}
    for key,value:=range body { if key!="game_id" && key!="resolved_token" { out[key]=value } }
    if _,ok:=body["resolved_token"]; ok { out["confirmation_token_present"]=true }
    return out
}
type v17InputOutcome struct {
    First,Confirmed map[string]any
    FirstError,ConfirmedError *gameapi.Error
}
// Constructed, single-input service probe. The four target contexts cover the
// actual warm cases and both owner-as-target wrong-win boundaries. No selector
// claim follows from these probes. Session tokens are omitted only from equality.
func v17InputProbe(p *v17Proof,s *Service,target,text string) v17InputOutcome {
    v:=s.build(target,"usor",nil,"")
    id,err:=s.store.Create(v); p.need(err==nil,"create finite input probe")
    defer s.store.Delete(id)
    body,apierr:=s.Guess(id,text,"")
    outcome:=v17InputOutcome{First:v17SanitizeBody(body),FirstError:apierr}
    if apierr!=nil { p.need(v.Attempts==0 && len(v.Guesses)==0 && !v.Won,"invalid input charged attempt") }
    if apierr==nil {
        won,_:=body["won"].(bool)
        for _,field:=range []string{"target","score","share","curated_id"} {
            if !won { _,leak:=body[field]; p.need(!leak,"unearned private field: "+field) }
        }
        if ok,_:=body["ok"].(bool); !ok {
            p.need(v.Attempts==0 && len(v.Guesses)==0 && !v.Won,"unknown or confirmation request charged attempt")
        }
        if labels,ok:=body["suggestions"].([]string); ok {
            for _,label:=range labels {
                owner:=s.g.Resolve(label)
                p.need(owner=="" || s.feedbackAnchor(owner,target,true)!=target,"suggestion leaked native target anchor")
                for _,term:=range s.data.Terms { if term.Label==label { p.need(s.feedbackAnchor(s.projectionAnchor(&term,target),target,false)!=target,"suggestion leaked projected target anchor") } }
            }
        }
        if needed,_:=body["needs_confirmation"].(bool); needed {
            token,ok:=body["resolved_token"].(string); p.need(ok && token!="","confirmation token missing")
            p.need(body["resolved_label"]!=s.g.Label(target),"confirmation leaked target label")
            confirmed,e:=s.Guess(id,text,token)
            outcome.Confirmed=v17SanitizeBody(confirmed); outcome.ConfirmedError=e
            p.need(e==nil,"valid confirmation rejected")
        }
        if v.Won { p.need(s.g.Resolve(text)==target || s.g.ResolveFuzzy(text)==target,"wrong-owner win") }
    }
    return outcome
}

            eligible,storedWeight=native.PilotEligible,native.SelectionWeight
        } else { p.need(item.Status=="pending","approved source record missing from sealed pool") }
        ap,bp:=v17FreshProfile(a,target),v17FreshProfile(b,target)
        av,bv:=a.build(target,item.Difficulty,nil,""),b.build(target,item.Difficulty,nil,"")
        oldView,newView:=v17ProfileView(ap),v17ProfileView(bp)
        p.same("full_profiles",item.ID,oldView,newView)
        oldWin,newWin:=v17InputProbe(p,a,target,a.g.Label(target)),v17InputProbe(p,b,target,b.g.Label(target))
        p.need(oldWin.FirstError==nil && newWin.FirstError==nil && oldWin.First["won"]==true && newWin.First["won"]==true,"canonical target does not win: "+item.ID)
        oldAnswer,oldOK:=oldWin.First["target"].(map[string]any); newAnswer,newOK:=newWin.First["target"].(map[string]any)
        p.need(oldOK && newOK && oldAnswer["id"]==target && newAnswer["id"]==target,"actual canonical winner identity drift")
        p.same("actual_canonical_winners",item.ID,oldWin,newWin)
        hashes:=[4]hash.Hash{sha256.New(),sha256.New(),sha256.New(),sha256.New()}
        cold:=make([]v17FeedbackRow,0,len(ids))
        for _,id:=range ids {
            old,new:=v17Feedback(a,av,id,nil),v17Feedback(b,bv,id,nil)
            p.same("native_feedback_pairs",item.ID+"/"+id,old,new)
            p.same("anchor_policy_pairs",item.ID+"/"+id,[]string{a.feedbackAnchor(id,target,false),a.feedbackAnchor(id,target,true)},[]string{b.feedbackAnchor(id,target,false),b.feedbackAnchor(id,target,true)})
            v17HashRow(hashes[0],old); v17HashRow(hashes[1],new)
            if id!=target { cold=append(cold,old) }
        }
        for _,key:=range a.projectionKeys {
            old,new:=v17Feedback(a,av,"",a.projections[key]),v17Feedback(b,bv,"",b.projections[key])
            p.same("projection_feedback_pairs",item.ID+"/"+key,old,new)
            p.need(old.WinnerID=="" && new.WinnerID=="","projection gained winner identity")
            v17HashRow(hashes[2],[]any{key,old}); v17HashRow(hashes[3],[]any{key,new})
        }
        // Three farthest baseline native feedback guesses, shared unchanged IDs.
        // Unreachable feedback uses999; ties: rank descending,distance descending,ID.
        sort.Slice(cold,func(i,j int)bool { x,y:=cold[i],cold[j]; if x.Rank!=y.Rank{return x.Rank>y.Rank}; if x.Distance!=y.Distance{return x.Distance>y.Distance}; return x.ID<y.ID })
        p.need(len(cold)>=3,"three old nonwinning guesses unavailable")
        coldIDs:=[]string{cold[0].ID,cold[1].ID,cold[2].ID}
        oldClue,newClue:=v17ClueProbe(p,a,target,item.Difficulty,coldIDs),v17ClueProbe(p,b,target,item.Difficulty,coldIDs)
        p.same("clue_profiles",item.ID,oldClue,newClue)
        p.emit(map[string]any{"kind":"target_receipt","pack_id":item.ID,"target":target,"status":item.Status,"pilot_eligible":eligible,"selection_weight":storedWeight,"native_count":len(ids),"projection_count":len(a.projectionKeys),"reachable_before":ap.Reachable,"reachable_after":bp.Reachable,"profile_before_sha256":v17Digest(oldView),"profile_after_sha256":v17Digest(newView),"native_before_sha256":fmt.Sprintf("%x",hashes[0].Sum(nil)),"native_after_sha256":fmt.Sprintf("%x",hashes[1].Sum(nil)),"projection_before_sha256":fmt.Sprintf("%x",hashes[2].Sum(nil)),"projection_after_sha256":fmt.Sprintf("%x",hashes[3].Sum(nil)),"cold_guess_ids":coldIDs,"clue_before":oldClue,"clue_after":newClue})
        p.counts["pack_targets"]++
    }
    p.need(statuses["approved"]==263 && statuses["pending"]==2 && len(statuses)==2,"263approved/2pending retained")
    for _,s:=range []*Service{a,b} { s.profiles=&profileCache{values:map[string]*profile{}} }
    p.stage="mined_eligibility"
    miningHashes:=[2]hash.Hash{sha256.New(),sha256.New()}
    highA,highB:=a.g.BySalience(.6,true),b.g.BySalience(.6,true)
    lowA,lowB:=a.g.BySalience(0,false),b.g.BySalience(0,false)
    p.same("mining_salience_order","easy",highA,highB); p.same("mining_salience_order","hard",lowA,lowB)
    for _,id:=range ids {
        old,new:=v17Mining(a,id),v17Mining(b,id)
        p.same("mined_targets",id,old,new)
        v17HashRow(miningHashes[0],[]any{id,old}); v17HashRow(miningHashes[1],[]any{id,new})
    }
    p.emit(map[string]any{"kind":"mining_receipt","targets":len(ids),"before_sha256":fmt.Sprintf("%x",miningHashes[0].Sum(nil)),"after_sha256":fmt.Sprintf("%x",miningHashes[1].Sum(nil)),"policy":"CreateWithExclusions: preferred reachable>=120 and responsive distances1..5>=40; fallback reachable>=120; final largest-component fallback subject to category120 refusal. Easy salience>=.6 when nonempty; hard lower half of ascending salience. All IDs measured; incoming>=5 is a separate unchanged C3 numeric floor, never target approval."})
    p.stage="ordinary_weights"
    categories:=append([]string{""},before.CategoryOrder...)
    for _,cat:=range categories { for _,difficulty:=range []string{"usor","normal","greu"} {
        opts:=pack.PickOptions{Category:cat,Difficulty:difficulty,FilteredShelfWeights:true}
        old,new:=v17Weights(a,opts),v17Weights(b,opts)
        p.same("ordinary_weight_shelves",cat+"/"+difficulty,old,new)
        picks:=map[string][]string{}
        for _,seed:=range []int64{0,3,7,99} {
            x:=a.pack.PickSeeded("contexto",pyrandom.New(big.NewInt(seed)),opts); y:=b.pack.PickSeeded("contexto",pyrandom.New(big.NewInt(seed)),opts)
            xid,yid:="",""; if x!=nil{xid=x.ID}; if y!=nil{yid=y.ID}
            p.same("ordinary_seeded_choices",fmt.Sprintf("%s/%s/%d",cat,difficulty,seed),xid,yid)
            picks[fmt.Sprint(seed)]=[]string{xid,yid}
        }
        p.emit(map[string]any{"kind":"weight_shelf","category":cat,"difficulty":difficulty,"before":old,"after":new,"finite_seeded_choices":picks})
    } }
    p.need(p.counts["pack_targets"]==265 && p.counts["native_feedback_pairs"]==641035 && p.counts["projection_feedback_pairs"]==122960,"complete265target feedback matrix coverage")
    p.need(p.counts["old_normalized_identity"]==13514 && p.counts["projection_identity"]==464 && p.counts["mined_targets"]==2419,"complete identity/projection/mining coverage")
    p.need(p.counts["fuzzy_queries"]==48 && p.counts["service_input_pairs"]==192 && p.counts["clue_profiles"]==265 && p.counts["actual_canonical_winners"]==265 && p.counts["direct_alias_case_contexts"]==24,"complete fixed29+corpus19/input/clue/winner coverage")
    p.need(p.counts["other69_native_dispositions"]==69,"complete separate69 lexical classification")
    p.stage="complete"
    p.complete=true
    unexpected:=0; for _,n:=range p.differences {unexpected+=n}
    if unexpected!=0 { t.Errorf("%d unexpected preservation differences; exact rows retained in proof",unexpected) }
    t.Logf("research coverage complete:265 targets,641035 native pairs,122960 projection pairs,2419 mining profiles,13514 old keys; %d fuzzy and %d service-input changes require review",p.counts["fuzzy_query_deltas"],p.counts["service_input_deltas"])
}

type v17ClueResult struct { Before map[string]any; Category,Warmer map[string]any; CategoryError,WarmerError *gameapi.Error; Final map[string]any }
func v17ClueProbe(p *v17Proof,s *Service,target,difficulty string,ids []string) v17ClueResult {
    v:=s.build(target,difficulty,nil,"")
    for _,id:=range ids {
        row:=v17Feedback(s,v,id,nil)
        v.Attempts++; v.Order=append(v.Order,id)
        v.Guesses[id]=&record{ID:id,Label:row.Label,Distance:row.Distance,Closeness:row.Closeness,Rank:row.Rank,Temperature:row.Temperature,Anchor:row.Anchor,Attempt:v.Attempts}
    }
    sid,err:=s.store.Create(v); p.need(err==nil,"create constructed clue session"); defer s.store.Delete(sid)
    out:=v17ClueResult{Before:s.clueView(v)}
    body,e:=s.Clue(sid); out.Category=v17SanitizeBody(body); out.CategoryError=e
    body,e=s.Clue(sid); out.Warmer=v17SanitizeBody(body); out.WarmerError=e
    out.Final=s.clueView(v)
    p.need(!v.Won && len(v.Guesses)==3 && v.Attempts==3,"clue changed guesses or winner")
    if v.Warm!=nil { p.need(v.Warm.Label!=s.g.Label(target),"warmer clue exposed target") }
    return out
}
type v17MiningRow struct { Reachable,Responsive int; Preferred,Fallback,C3NumericFloor bool; Incoming []string; Category string; SalienceBits uint64 }
func v17Mining(s *Service,id string) v17MiningRow {
    dist:=s.g.DistancesToDense(id)
    row:=v17MiningRow{Incoming:[]string{},Category:s.g.Node(id).Category,SalienceBits:math.Float64bits(s.g.Salience(id))}
    for _,d:=range dist { if d>=0 {row.Reachable++}; if d>=1 && d<=5 {row.Responsive++} }
    seen:=map[string]bool{}
    for _,previous:=range s.g.PredecessorIDs(id) { if previous!=id && s.g.Exists(previous) && !seen[previous] {seen[previous]=true;row.Incoming=append(row.Incoming,previous)} }
    sort.Strings(row.Incoming)
    row.Preferred=row.Reachable>=120 && row.Responsive>=40; row.Fallback=row.Reachable>=120; row.C3NumericFloor=len(row.Incoming)>=5
    return row
}
type v17WeightRow struct { ID string; PilotScore,StoredWeight,EffectiveWeight int; Eligible bool; Status string }
func v17Weights(s *Service,opts pack.PickOptions) []v17WeightRow {
    pool:=s.pack.Pool("contexto",opts)
    eligible:=[]*content.PackItem{}; for _,v:=range pool {if v.PilotEligible{eligible=append(eligible,v)}}
    sort.Slice(eligible,func(i,j int)bool{if eligible[i].PilotScore!=eligible[j].PilotScore{return eligible[i].PilotScore>eligible[j].PilotScore};return eligible[i].ID<eligible[j].ID})
    effective:=map[string]int{}
    // Mirrors the exact pinned private pack.weights arithmetic for a complete
    // shelf receipt; PickSeeded above separately invokes the actual policy.
    for i,v:=range eligible {effective[v.ID]=5-min(4,5*i/len(eligible))}
    out:=[]v17WeightRow{}
    for _,v:=range pool {out=append(out,v17WeightRow{v.ID,v.PilotScore,v.SelectionWeight,effective[v.ID],v.PilotEligible,v.Status})}
    return out
}
