def browser_cases(log):
    cases = {}
    for line in log.splitlines():
        line = re.sub(r'\x1b\[[0-9;]*m', '', line)
        match = re.match(r'^\s+(ok|x)\s+\d+\s+\[([^]]+)\]\s+›\s+(?:frontend[\\/])?e2e[\\/](.+?\.spec\.mjs):\d+:\d+\s+›\s+(.*?)\s+\([^)]*\)\s*$', line)
        if match:
            status, project, file, title = match.groups()
            cases[f'[{project}] {file} > {title}'] = status
    return cases
full_cases, scope_cases = browser_cases(browser_log), browser_cases(scope_log)
