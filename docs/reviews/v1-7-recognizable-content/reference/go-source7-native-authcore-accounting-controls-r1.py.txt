import ast,pathlib,os,stat,tempfile,types,sys,unittest
from unittest import mock
sys.dont_write_bytecode=True
S=pathlib.Path(sys.argv.pop(1));W=S/'probe/native-authcore-race-compile-windows-r4.py'
node=next(x for x in ast.parse(W.read_text()).body if isinstance(x,ast.FunctionDef) and x.name=='count')
class Scan:
 def __init__(self,real,hook,entryhook):self.real=real;self.hook=hook;self.entryhook=entryhook
 def __call__(self,p):
  if self.hook:self.hook(pathlib.Path(p))
  outer=self;inner=self.real(p)
  class Context:
   def __enter__(self):
    entries=inner.__enter__()
    class Entry:
     def __init__(self,e):self.path=e.path;self.e=e
     def stat(self,**kw):
      if outer.entryhook:return outer.entryhook(self.e,kw)
      return self.e.stat(**kw)
    return (Entry(e) for e in entries)
   def __exit__(self,*args):return inner.__exit__(*args)
  return Context()
class Cases(unittest.TestCase):
 def setUp(self):
  self.tmp=tempfile.TemporaryDirectory(dir=S/'probe');self.base=pathlib.Path(self.tmp.name)
  self.root=self.base/'R';self.scratch=self.base/'S';self.cache=self.base/'cache';self.mod=self.base/'mod';self.run=self.scratch/'run'
  for p in (self.root,self.run/'tmp',self.cache,self.mod):p.mkdir(parents=True,exist_ok=True)
  for p,size in [(self.root/'source',7),(self.run/'tmp'/'temp',11),(self.cache/'cachefile',13),(self.mod/'module',17)]:p.write_bytes(b'x'*size)
  self.env={'os':types.SimpleNamespace(scandir=os.scandir),'stat':stat,'pathlib':pathlib,'root':self.root,'scratch':self.scratch,'run':self.run,'spec':{'linux_gocache':str(self.cache).replace('C:\\','/mnt/c/').replace('\\','/'),'windows_gomodcache':str(self.mod)},'carrier':types.SimpleNamespace(poll=lambda:None)}
  exec(compile(ast.Module(body=[node],type_ignores=[]),str(W),'exec'),self.env)
 def tearDown(self):self.tmp.cleanup()
 def count(self,hook=None,entryhook=None):
  self.env['os'].scandir=Scan(os.scandir,hook,entryhook);return self.env['count']()
 def disappearing_file(self,p):
  def f(e,kw):
   if pathlib.Path(e.path)==p:p.unlink();raise FileNotFoundError(str(p))
   return e.stat(**kw)
  return f
 def test_normal(self):self.assertEqual(self.count(),48)
 def test_vanished_temp_file(self):self.assertEqual(self.count(entryhook=self.disappearing_file(self.run/'tmp'/'temp')),37)
 def test_vanished_cache_file(self):self.assertEqual(self.count(entryhook=self.disappearing_file(self.cache/'cachefile')),35)
 def vanished_dir(self,parent):
  d=parent/'gone';d.mkdir()
  def hook(p):
   if p==parent:d.rmdir()
  # Inject directory removal only after its entry stat queued it.
  def entry(e,kw):
   info=e.stat(**kw)
   if pathlib.Path(e.path)==d:d.rmdir()
   return info
  self.assertEqual(self.count(entryhook=entry),48)
