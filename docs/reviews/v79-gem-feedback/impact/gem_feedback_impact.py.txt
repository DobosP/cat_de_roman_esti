MIERE = 'n_v24_food_breakfast_miere'
DULCEATA = 'n_v17gas_dulceata'
CLATITE = 'n_v3gas_clatite'
PROJECTION_SOURCE = ROOT / 'cat_de_roman_esti/wordgames/contexto_projection.py'
CONTEXTO_SOURCE = ROOT / 'cat_de_roman_esti/wordgames/contexto.py'
KG_PATH = ROOT / 'cat_de_roman_esti/fixtures/kg_sample.json'

def sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def strings(value):
    if isinstance(value, str):
        yield value
    elif isinstance(value, dict):
        for item in value.values():
            yield from strings(item)
    elif isinstance(value, list):
        for item in value:
            yield from strings(item)


def ids(value):
    if isinstance(value, dict):
        for key, item in value.items():
            if key == 'id':
                yield item
            yield from ids(item)
    elif isinstance(value, list):
        for item in value:
            yield from ids(item)


def public_subset(body: dict) -> dict:
    return {
        key: body.get(key)
        for key in (
            'ok', 'needs_confirmation', 'message', 'suggestions', 'attempts', 'won',
            'reachable_count', 'guess', 'guesses',
        )
        if key in body
    }


def git_source_sha(relative_path: str) -> str:
    content = subprocess.check_output(['git', 'show', f'HEAD:{relative_path}'], cwd=ROOT)
    return hashlib.sha256(content).hexdigest()


def capture(phase: str, output: Path, *, force_default_anchor: bool) -> None:
    import django
    django.setup()
    from django.test import Client
    from cat_de_roman_esti.wordgames import contexto
    from cat_de_roman_esti.wordgames.contexto_projection import (
        PROJECTION_NEIGHBORHOODS,
        PROJECTION_TERMS,
    )
    from cat_de_roman_esti.wordgames.packs import get_pack
    from cat_de_roman_esti.wordgames.service import get_service

    client = Client()
    svc = get_service()
    pack = get_pack()
    if force_default_anchor:
        contexto._projection_anchor_id = lambda _svc, term, _target: term.anchor_id

    def post_for(target: str, text: str, *, difficulty='usor', category='gastronomie'):
        session = contexto._build_session(target, difficulty, None, category=category)
        game_id = contexto.store.create(session)
        try:
            response = client.post(
                f'/api/wordgames/contexto/games/{game_id}/guess',
                {'text': text}, content_type='application/json',
            )
            body = response.json()
            target_label = svc.label(target)
            return {
                'status': response.status_code,
                'body': public_subset(body),
                'target_hidden': (
                    'target' not in body and 'solution' not in body
                    and target not in set(ids(body))
                    and target_label not in set(strings(body))
                ),
            }
        finally:
            if not contexto.store.delete(game_id):
                raise RuntimeError(f'temporary session was not deleted: {game_id}')

    approved = pack.pool('contexto')
    sweep = []
    for item in approved:
        target = str(item.payload['target'])
        result = post_for(
            target, GEM_KEY, difficulty=item.difficulty, category=item.category,
        )
        sweep.append({
            'pack_id': item.id,
            'target_id': target,
            'target_label': svc.label(target),
            'category': item.category,
            'difficulty': item.difficulty,
            'pilot_eligible': item._pilot_eligible,
            **result,
        })

    fixed = {}
    fixed_targets = (
        MIERE, DULCEATA, 'n_gas_papanasi', 'n_v11gas_magiun_topoloveni',
        'n_v18gas_socata', 'n_v2gas_conserve_iarna', CLATITE, 'n_v4gas_fruct',
    )
    for target in fixed_targets:
        fixed[target] = {
            'gem': post_for(target, 'gem'),
            'gem_typo': post_for(target, 'gemm'),
            'exact': post_for(target, svc.label(target)),
        }

    mappings = {
        term.key: {
            'surface': term.surface,
            'anchor_id': term.anchor_id,
            'domain': term.domain,
            'rank_penalty': term.rank_penalty,
            'mapping_kind': term.mapping_kind,
            'public_id': term.public_id,
        }
        for term in PROJECTION_TERMS
    }
    document = {
        'meta': {
            'phase': phase,
            'behavior_override': 'force-default-projection-anchor' if force_default_anchor else None,
            'git_head': subprocess.check_output(
                ['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True,
            ).strip(),
            'projection_source_sha256': sha256(PROJECTION_SOURCE),
            'contexto_source_sha256': sha256(CONTEXTO_SOURCE),
            'baseline_projection_source_sha256': git_source_sha(
                'cat_de_roman_esti/wordgames/contexto_projection.py'
            ),
            'baseline_contexto_source_sha256': git_source_sha(
                'cat_de_roman_esti/wordgames/contexto.py'
            ),
            'kg_sha256': sha256(KG_PATH),
            'pack_sha256': sha256(PACK_PATH),
            'rankings_sha256': sha256(RANKINGS_PATH),
            'derived_sha256': sha256(DERIVED_PATH),
            'approved_targets': len(approved),
            'eligible_targets': sum(item._pilot_eligible for item in approved),
            'projection_terms': len(mappings),
        },
        'projection_mappings': mappings,
        'projection_neighborhoods': {} if force_default_anchor else {
            key: {'anchor_id': value.anchor_id, 'min_strength': value.min_strength}
            for key, value in sorted(PROJECTION_NEIGHBORHOODS.items())
        },
        'approved_target_gem_feedback': sweep,
        'fixed_target_public_probes': fixed,
    }
    output.write_text(
        json.dumps(document, ensure_ascii=False, indent=2) + '\n', encoding='utf-8'
    )
    print(json.dumps(document['meta'], ensure_ascii=False, indent=2))


def compare(before_path: Path, after_path: Path, output: Path) -> None:
    before = json.loads(before_path.read_text(encoding='utf-8'))
    after = json.loads(after_path.read_text(encoding='utf-8'))
    bm, am = before['projection_mappings'], after['projection_mappings']
    mapping_changes = {
        key: {'before': bm.get(key), 'after': am.get(key)}
        for key in sorted(set(bm) | set(am)) if bm.get(key) != am.get(key)
    }
    bs = {row['pack_id']: row for row in before['approved_target_gem_feedback']}
    a_s = {row['pack_id']: row for row in after['approved_target_gem_feedback']}
    if set(bs) != set(a_s):
        raise RuntimeError('approved target batch changed')
    transitions = Counter()
    changed = []
    newly_warm = []
    warm = {'Fierbinte', 'Cald', 'Caldut'}
    hot = {'Fierbinte', 'Cald'}
    for pack_id in sorted(bs):
        old, new = bs[pack_id], a_s[pack_id]
        old_guess = old['body']['guess']
        new_guess = new['body']['guess']
        transitions[(old_guess['temperature'], new_guess['temperature'])] += 1
        if old_guess != new_guess:
            row = {
                'pack_id': pack_id,
                'target_id': old['target_id'],
                'target_label': old['target_label'],
                'category': old['category'],
                'pilot_eligible': old['pilot_eligible'],
                'before': old_guess,
                'after': new_guess,
            }
            changed.append(row)
            if new_guess['temperature'] in warm and old_guess['temperature'] not in warm:
                newly_warm.append(row)
    fixed_before = before['fixed_target_public_probes']
    fixed_after = after['fixed_target_public_probes']
    result = {
        'bindings': {'before': before['meta'], 'after': after['meta']},
        'unchanged_artifact_hashes': {
            key: before['meta'][key] == after['meta'][key]
            for key in ('kg_sha256', 'pack_sha256', 'rankings_sha256', 'derived_sha256')
        },
        'mapping_changes': mapping_changes,
        'all_projection_mappings_exact': not mapping_changes,
        'projection_neighborhood_changes': {
            'before': before.get('projection_neighborhoods', {}),
            'after': after.get('projection_neighborhoods', {}),
        },
        'bounded_policy_is_expected': (
            not mapping_changes
            and after.get('projection_neighborhoods') == {
                GEM_KEY: {'anchor_id': DULCEATA, 'min_strength': 0.60}
            }
            and am[GEM_KEY]['anchor_id'] == MIERE
            and am[GEM_KEY]['public_id'] == EXPECTED_PUBLIC_ID
        ),
        'approved_target_counts': {
            'total': len(bs),
            'eligible': sum(row['pilot_eligible'] for row in bs.values()),
            'feedback_changed': len(changed),
            'eligible_feedback_changed': sum(row['pilot_eligible'] for row in changed),
        },
        'temperature_transitions': [
            {'before': pair[0], 'after': pair[1], 'count': count}
            for pair, count in sorted(transitions.items())
        ],
        'newly_warm': newly_warm,
        'newly_hot_cross_category': [
            row for row in changed
            if row['category'] != 'gastronomie'
            and row['after']['temperature'] in hot
            and row['before']['temperature'] not in hot
        ],
        'changed_feedback': changed,
        'fixed_target_public_probes': {
            target: {'before': fixed_before[target], 'after': fixed_after[target]}
            for target in fixed_before
        },
        'all_sweep_targets_hidden': all(
            row['target_hidden'] for row in bs.values()
        ) and all(row['target_hidden'] for row in a_s.values()),
    }
    output.write_text(json.dumps(result, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
    print(json.dumps({
        'mapping_changes': list(mapping_changes),
        'bounded_policy_expected': result['bounded_policy_is_expected'],
        'counts': result['approved_target_counts'],
        'transitions': result['temperature_transitions'],
        'newly_warm': len(newly_warm),
        'newly_hot_cross_category': len(result['newly_hot_cross_category']),
        'all_hidden': result['all_sweep_targets_hidden'],
        'artifacts_unchanged': result['unchanged_artifact_hashes'],
    }, ensure_ascii=False, indent=2))


def main() -> None:
    parser = argparse.ArgumentParser()
    sub = parser.add_subparsers(dest='command', required=True)
    cap = sub.add_parser('capture')
    cap.add_argument('--phase', required=True)
    cap.add_argument('--out', type=Path, required=True)
    cap.add_argument('--force-default-anchor', action='store_true')
    comp = sub.add_parser('compare')
    comp.add_argument('--before', type=Path, required=True)
    comp.add_argument('--after', type=Path, required=True)
    comp.add_argument('--out', type=Path, required=True)
    args = parser.parse_args()
    args.out.parent.mkdir(parents=True, exist_ok=True)
    if args.command == 'capture':
        capture(args.phase, args.out, force_default_anchor=args.force_default_anchor)
    else:
        compare(args.before, args.after, args.out)


if __name__ == '__main__':
    main()
