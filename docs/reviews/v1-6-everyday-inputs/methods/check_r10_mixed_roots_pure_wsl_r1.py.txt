def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def guard():
    if time.monotonic() - STARTED >= 10:
        raise TimeoutError('pure fixture10seconds')
    if resource.getrusage(resource.RUSAGE_SELF).ru_maxrss * 1024 > 2 * GIB:
        raise RuntimeError('pure fixture RSS2GiB')


def chain(path):
    for value in reversed([Path(path), *Path(path).parents]):
        if not stat.S_ISDIR(value.lstat().st_mode) or value.resolve(strict=True) != value:
            raise ValueError('fixture ancestor redirected/non-directory')


def write(path, blob):
    guard()
    if len(blob) > 128 << 10:
        raise ValueError('fixture image128KiB')
    with Path(path).open('xb') as f:
        f.write(blob)
        f.flush()
        os.fsync(f.fileno())


def case(name, function, wanted=None):
    guard()
    if wanted is None:
        value = function()
        if value is not True:
            raise RuntimeError(name + ': expected True')
    else:
        try:
            function()
        except wanted:
            pass
        else:
            raise RuntimeError(name + ': expected ' + wanted.__name__)
    CASES.append(name)


def synthetic_reply(m, requested, receipt):
    quota = {'minimum_bytes': 1 << 20, 'maximum_bytes': 64 << 20, 'enforcement_flags': 6}
    ids = {role: {'pid': 100 + i, 'creation_time_100ns': 200 + i} for i, role in enumerate(('parent', 'launcher', 'worker'))}
    proofs = {}
    for role, scope in (('parent', 'parent-own-current-process'), ('launcher', 'exact-owned-windows-job'), ('worker', 'exact-owned-windows-job'), ('worker_self', 'worker-own-current-process')):
        proof = dict(quota, **ids['worker' if role == 'worker_self' else role], ownership_scope=scope,
                     hard_max_verified=True, identity_verified_before_quota=True, before_traversal=True,
                     quota_call_succeeded=True, job_membership_verified_before_quota=True,
                     exact_owned_job_membership_verified_before_quota=True)
        if role == 'launcher':
            proof['enforced_while_suspended'] = True
        proofs[role] = proof
    return dict(ids, schema='v1-6-windows-fresh-size-helper-r2', nonce=receipt.stem,
                roots=requested, receipt_path=m.windows_path(receipt), status='completed', regular_bytes=37,
                worker_exit_confirmed=True, launcher_exit_confirmed=True, job_empty=True,
                elapsed_seconds=.01, peak_job_memory_bytes=1, peak_working_set_bytes=1,
                helper_sha256=m.WINDOWS_HELPER_SHA, interpreter_sha256=m.WINDOWS_PYTHON_SHA,
                python_version='3.11.15', deferred=[], hard_working_set_proofs=proofs,
                launcher_working_set_before_traversal=quota, parent_working_set_at_completion=quota,
                worker_working_set_at_completion=quota, job_setup={'configured': True, 'limit_flags': 8968},
                job_memory_limit_bytes=64 << 20, process_memory_limit_bytes=64 << 20,
                job_active_process_limit=2, job_closed_confirmed=True,
                synthetic_fixture_reply_not_OS_evidence=True)
